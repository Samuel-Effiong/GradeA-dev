#!/bin/bash
# AI-call record, slice 0. Usage: bash run_ai_call_record_0_gate.sh <tip> 1
# STEP=1 (6G, one grant): 0 the red run (the four new test modules against the code as at the tests-only commit
#   $RED: vote.py, step_run.py and students/step_label.py taken away, grading_run.py taken back; own DB),
#   1a makemigrations --check, 1b the seam check (git diff of ai_processor/services.py and of slice C's test
#   modules against the base is EMPTY), 1 changed modules + slice C's label modules + near modules + guards,
#   2 the 33 mutants (rule 17; own DB).
# The regression is 0b's one full run on the frozen tip (SM ruling); it is not in this script.
# A trap on EXIT restores the tree. Rules 12/13/16. RULE 18: every run writes stdout+stderr straight to a file,
# stdin from /dev/null; nothing is piped.
set -u
TIP=$1; STEP=${2:-1}; RED=c67bcf17; BASE=75d82620
cd /home/bond-servant-in-training/Documents/Projects/Grade-Automator-Plus-epic-i-ai-call-record-0 || exit 1
source /home/bond-servant-in-training/Documents/Virtualenvs/AutoGrader_env/bin/activate
E=docs/evidence/epic-i-ai-call-record-0
# Rule 16 (revised): the whole script runs inside ONE outer systemd-inhibit, started as
#   systemd-inhibit --what=idle:sleep:handle-lid-switch --who=GAP --why=GAP-test-run --mode=block bash <script> <tip> 1
# so there is no inner inhibit; the memory scope and the timeout stay per test command.
CAP="systemd-run --user --scope -q -p MemoryMax=6G -p MemorySwapMax=0"
export PYTHONFAULTHANDLER=1
export EXEMPT_EMAIL_DOMAINS=
[ "$(git rev-parse HEAD)" = "$(git rev-parse $TIP)" ] || { echo "not at $TIP"; exit 1; }
git diff --quiet HEAD || { echo "tree not clean"; exit 1; }
[ "$STEP" = 1 ] || { echo "only step 1 exists"; exit 1; }
NEWFILES="ai_processor/vote.py ai_processor/step_run.py students/step_label.py"
CODE="ai_processor/grading_run.py $NEWFILES"
restore_tree() { git checkout HEAD -- $CODE 2>/dev/null; find ai_processor students -name __pycache__ -type d -prune -exec rm -rf {} + 2>/dev/null; }
trap restore_tree EXIT
summary() { grep -E "^(Ran |OK|FAILED \(|(FAIL|ERROR): )" "$1" | cut -c1-200 | head -${2:-12}; }
dropmut() { python manage.py shell --settings=settings_worktree_mut -c "
from django.conf import settings
from django.db import connection
name = settings.DATABASES['default']['TEST']['NAME']
with connection._nodb_cursor() as c:
    c.execute('DROP DATABASE IF EXISTS \"%s\"' % name)
print('dropped', name)" < /dev/null > $E/.dropmut.out 2>&1; tail -1 $E/.dropmut.out; rm -f $E/.dropmut.out; }
RED_MODULES="ai_processor.tests_vote_helper ai_processor.tests_step_run ai_processor.tests_step_run_scope students.tests_step_label_words"
PIN="ai_processor.tests_grading_run_label ai_processor.tests_grading_run_pipeline ai_processor.tests_grading_run_checker students.tests_grading_label_written students.tests_grading_label_routes students.tests_grading_label_not_exposed students.tests_grading_label_migration students.tests_grading_label_fields students.tests_grading_label_end_to_end ai_processor.tests_grading_cache ai_processor.tests_grading_config"
NEAR="ai_processor.tests_second_opinion_pipeline ai_processor.tests_second_opinion_selection ai_processor.tests_grading_pipeline ai_processor.tests_grading_completeness ai_processor.tests_grading_arithmetic ai_processor.tests_ai_call_trace ai_processor.tests_reply_repeated_evaluation ai_processor.tests_second_opinion_error_code students.tests_s6d_provider_failure students.tests_app_boundaries students.tests_grading_idempotency assignments.tests_grading_audit_events audit.tests_metrics audit.tests_history"
GUARDS="AutoGrader.tests_no_wildcard_invalidation AutoGrader.tests_cache_invalidation_coverage AutoGrader.tests_reason_codes AutoGrader.tests_migration_rollback_defaults AutoGrader.tests_redis_test_isolation AutoGrader.tests_beat_health AutoGrader.tests_beat_locks AutoGrader.tests_management_commands_are_commands AutoGrader.tests_error_messages AutoGrader.tests_no_pii_in_logs AutoGrader.tests_log_scrubbing AutoGrader.tests_sentry_scrubbing AutoGrader.tests_redis_hygiene_databases AutoGrader.tests_redis_hygiene AutoGrader.tests_patient_test_stream AutoGrader.tests_pool_worker_sigterm AutoGrader.tests_codederror_serialization AutoGrader.tests_student_feedback_guard AutoGrader.tests_submission_audience_guard AutoGrader.tests_student_classmates_guard audit.tests_history_guard audit.tests_route_coverage audit.tests_sweep_beat_lock"
echo "load at start: $(cat /proc/loadavg)"
echo "=== 0. the red run: the four new test modules against the code as at $RED (own DB) $(date +%T)"
git checkout $RED -- ai_processor/grading_run.py
rm -f $NEWFILES
find ai_processor students -name __pycache__ -type d -prune -exec rm -rf {} +
eval "PYTHONDONTWRITEBYTECODE=1 $CAP nice -n 10 timeout -k 60 1800 python manage.py test --noinput --verbosity 2 --settings=settings_worktree_mut $RED_MODULES" < /dev/null > $E/red_run_code_as_at_red_commit.txt 2>&1
RED_EXIT=$?
echo "red run exit=$RED_EXIT (non-zero expected)"
# Step 0 is compared with what was written (house rule): the written expectation is that ALL FOUR modules fail
# to LOAD because the modules they import did not exist yet. Halt (exit 4) unless: the exit is non-zero; no output
# line starts with "OK"; the run says "Ran 4 tests" and "FAILED (errors=4)"; each of the four modules is named in a
# load error (the runner names a failed module by its LAST component: "ERROR: tests_vote_helper
# (unittest.loader._FailedTest.tests_vote_helper)" and "ImportError: Failed to import test module: tests_vote_helper",
# which I first wrote wrongly with the dotted name: the first run of this script stopped on that); and exactly four
# lines give the cause: the missing new module (ai_processor.vote, ai_processor.step_run, students.step_label).
RF=$E/red_run_code_as_at_red_commit.txt
RED_OK=1
[ "$RED_EXIT" -ne 0 ] || RED_OK=0
grep -qE "^OK" $RF && RED_OK=0
grep -qE "^Ran 4 tests" $RF || { echo "red run: not 'Ran 4 tests'"; RED_OK=0; }
grep -qE "^FAILED \(errors=4\)$" $RF || { echo "red run: not 'FAILED (errors=4)'"; RED_OK=0; }
for M in $RED_MODULES; do
    S=${M##*.}
    grep -qE "^ERROR: $S \(unittest.loader._FailedTest.$S\)$" $RF || { echo "red run: $S is not named in a load error"; RED_OK=0; }
    grep -qE "^ImportError: Failed to import test module: $S$" $RF || { echo "red run: $S has no 'Failed to import' line"; RED_OK=0; }
done
CAUSES=$(grep -cE "^(ModuleNotFoundError: No module named 'ai_processor.vote'|ImportError: cannot import name '(step_run|step_label)' from '(ai_processor|students)' )" $RF)
[ "$CAUSES" = 4 ] || { echo "red run: $CAUSES cause lines naming the missing new modules (4 expected)"; RED_OK=0; }
if [ "$RED_OK" != 1 ]; then
    echo "STOP (exit 4): the red run is not what was written (all four modules fail to load)"
    git checkout HEAD -- $CODE
    find ai_processor students -name __pycache__ -type d -prune -exec rm -rf {} +
    exit 4
fi
echo "red run is as written: non-zero exit, no OK line, Ran 4 tests, FAILED (errors=4), all four modules failed to load for the missing new modules"
git checkout HEAD -- $CODE
find ai_processor students -name __pycache__ -type d -prune -exec rm -rf {} +
git diff --quiet HEAD -- . ":!$E" && [ -z "$(git status --porcelain -- . ":!$E" | grep -v settings_worktree_mut.py)" ] && echo "source restored" || { echo "STOP: not restored"; exit 3; }
grep -E "^(Ran |OK|FAILED \\()" $E/red_run_code_as_at_red_commit.txt | cut -c1-200
dropmut
echo "=== 1a. makemigrations --check $(date +%T)"
python manage.py makemigrations --check --dry-run --settings=settings_worktree < /dev/null > $E/makemigrations_check.txt 2>&1; MM=$?; echo "makemigrations exit=$MM (0 = no changes)"; tail -1 $E/makemigrations_check.txt
[ "$MM" = 0 ] || { echo "STOP: makemigrations reports changes"; exit 2; }
echo "=== 1b. the seam: execute_graded_task, __ai_model and slice C's test modules are untouched"
git diff $BASE HEAD --stat -- ai_processor/services.py > $E/seam_services_diff.txt 2>&1
git diff $BASE HEAD --stat -- ai_processor/tests_grading_run_label.py ai_processor/tests_grading_run_pipeline.py ai_processor/tests_grading_run_checker.py students/tests_grading_label_written.py students/tests_grading_label_routes.py students/tests_grading_label_not_exposed.py students/tests_grading_label_migration.py students/tests_grading_label_fields.py students/tests_grading_label_end_to_end.py > $E/slice_c_tests_diff.txt 2>&1
echo "services.py diff bytes: $(wc -c < $E/seam_services_diff.txt) (0 expected); slice C tests diff bytes: $(wc -c < $E/slice_c_tests_diff.txt) (0 expected)"
[ ! -s $E/seam_services_diff.txt ] && [ ! -s $E/slice_c_tests_diff.txt ] || { echo "STOP: seam or slice C tests touched"; exit 2; }
echo "=== 1. changed modules + slice C pins + near modules + guards at $(git rev-parse --short=8 HEAD) $(date +%T)"
eval "$CAP nice -n 10 timeout -k 60 3000 python manage.py test --noinput --verbosity 2 --settings=settings_worktree $RED_MODULES $PIN $NEAR $GUARDS" < /dev/null > $E/modules_and_guards.txt 2>&1
echo "exit=$? $(date +%T)"; summary $E/modules_and_guards.txt
if grep -qE "^FAILED \\(" $E/modules_and_guards.txt || ! grep -qE "^OK" $E/modules_and_guards.txt; then echo "STOP: not green"; exit 2; fi
echo "=== 2. mutants: 33 (own DB, rules 17 and 18) $(date +%T)"
eval "MUT_SETTINGS=settings_worktree_mut MUT_RESULTS=$E/mutation_results.json MUT_LOGS=$E/mutant_logs $CAP nice -n 10 timeout -k 60 3400 python $E/mutate.py" < /dev/null > $E/mutation_log.txt 2>&1
echo "mutants exit=$? $(date +%T)"; cut -c1-260 $E/mutation_log.txt
git diff --quiet HEAD -- . ":!$E" && echo "source clean after mutants" || echo "STOP: source not clean after mutants"
dropmut
echo "load at end: $(cat /proc/loadavg)"
