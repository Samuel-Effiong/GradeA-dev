def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def git_show(rel):
    return subprocess.run(
        ["git", "show", f"{TIP}:{rel}"], cwd=WT, check=True, capture_output=True, text=True
    ).stdout


def run_tests(labels, log_path):
    cmd = [
        "python", "manage.py", "test", *labels,
        "--settings=settings_worktree", "--noinput", "--verbosity=2",
    ]
    with log_path.open("w") as fh:
        rc = subprocess.run(cmd, cwd=WT, stdout=fh, stderr=subprocess.STDOUT).returncode
    return rc, log_path.read_text(errors="replace")


def preflight():
    head = subprocess.run(
        ["git", "rev-parse", "--short=8", "HEAD"], cwd=WT, check=True, capture_output=True, text=True
    ).stdout.strip()
    dirty = subprocess.run(
        ["git", "status", "--porcelain"], cwd=WT, check=True, capture_output=True, text=True
    ).stdout.strip()
    if head != TIP or dirty:
        raise SystemExit(f"tree not frozen at {TIP} (HEAD {head}, dirty={bool(dirty)})")


def main():
    preflight()
    OUT.mkdir(parents=True, exist_ok=True)
    baseline_log = OUT / "baseline.log"
    base_labels = sorted({l for m in MUTANTS.values() for l in m["labels"]})
    rc, text = run_tests(base_labels, baseline_log)
    print(f"baseline: exit={rc}")
    if rc != 0:
        raise SystemExit("baseline is not green; aborting before mutating anything")

    results = {}
    for name, spec in MUTANTS.items():
        rel = spec["file"]
        path = WT / rel
        pre_sha = sha(path)
        source = path.read_text()
        count = source.count(spec["old"])
        if count != 1:
            raise SystemExit(f"{name}: anchor found {count} times in {rel}, expected 1")
        path.write_text(source.replace(spec["old"], spec["new"]))
        for cache in (WT / Path(rel).parent).rglob("__pycache__"):
            pass
        log_path = OUT / f"{name}.log"
        try:
            rc, text = run_tests(spec["labels"], log_path)
        finally:
            restored = git_show(rel)
            path.write_text(restored)
            post_sha = sha(path)
            ok = post_sha == pre_sha
            if not ok:
                raise SystemExit(f"{name}: RESTORE FAILED sha mismatch on {rel}")
        ran = "Ran " in text
        failed = rc != 0
        results[name] = (rc, ran, failed)
        print(f"{name}: exit={rc} ran={ran} restore_sha_ok={ok}")

    print("\nSummary:")
    for name, (rc, ran, failed) in results.items():
        verdict = "KILLED" if (ran and failed) else "SURVIVED (test suite stayed green!)"
        print(f"  {name}: {verdict} (exit={rc})")


if __name__ == "__main__":
    sys.exit(main())
