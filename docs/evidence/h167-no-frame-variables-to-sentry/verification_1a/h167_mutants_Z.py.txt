"""1a's mutants Z1 to Z5 for H-167 (rule 19: each probe of mine is seen red).
Usage, in the checkout: python -B h167_mutants_Z.py <Z1|Z2|Z3|Z4|Z5>. Applied
in place; restore with git checkout. Expected failing tests: h167_expected_kills.txt.

Z1: the init call does not pass include_local_variables.
Z2: the init call lets medium request bodies through.
Z3: the hooks leave an event's contexts alone.
Z4: the pattern does not know the fourth name.
Z5: the hooks leave an event's exception alone.
"""

import ast
import sys

MUTANTS = {
    "Z1": (
        "AutoGrader/settings.py",
        "            include_local_variables=False,\n",
        "",
    ),
    "Z2": (
        "AutoGrader/settings.py",
        '            max_request_body_size="never",\n',
        '            max_request_body_size="medium",\n',
    ),
    "Z3": (
        "AutoGrader/sentry_scrubbing.py",
        '    "contexts",\n',
        "",
    ),
    "Z4": (
        "AutoGrader/log_scrubbing.py",
        '_NAMED_VALUE = re.compile(r"(?i)(password|passwd|secret|token)=[^\\n]+")\n',
        '_NAMED_VALUE = re.compile(r"(?i)(password|passwd|secret)=[^\\n]+")\n',
    ),
    "Z5": (
        "AutoGrader/sentry_scrubbing.py",
        '    "exception",\n',
        "",
    ),
}


def main():
    name = sys.argv[1]
    path, old, new = MUTANTS[name]
    with open(path, encoding="utf-8") as handle:
        source = handle.read()
    count = source.count(old)
    if count != 1:
        print(f"ABORT {name}: anchor found {count} times in {path}")
        sys.exit(7)
    mutated = source.replace(old, new, 1)
    ast.parse(mutated)
    if "--check" in sys.argv:
        print(f"checked {name}: anchor once in {path}, mutant parses")
        return
    with open(path, "w", encoding="utf-8") as handle:
        handle.write(mutated)
    print(f"applied {name} to {path}")


if __name__ == "__main__":
    main()
