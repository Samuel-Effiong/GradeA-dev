def sha(b): return hashlib.sha256(b).hexdigest()

def run(labels, name):
    LOGS.mkdir(parents=True, exist_ok=True); log = LOGS / name
    with open(log, "wb") as fh:
        p = subprocess.run([sys.executable, "-B", "manage.py", "test", *labels, "--settings=settings_worktree", "--noinput", "--keepdb", "-v1"],
                           stdout=fh, stderr=subprocess.STDOUT, env=ENV, stdin=subprocess.DEVNULL)
    o = log.read_text(errors="replace")
    failing = sorted(set(re.findall(r"^(?:FAIL|ERROR): (\w+)", o, re.M)))
    ran = next((l for l in o.splitlines() if l.startswith("Ran ")), "Ran ?")
    summ = next((l for l in o.splitlines() if l.startswith(("OK", "FAILED"))), f"? rc={p.returncode}")
    return p.returncode, ran, summ, failing, o
