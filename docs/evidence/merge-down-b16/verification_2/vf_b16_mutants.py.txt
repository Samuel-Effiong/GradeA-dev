ENV = {**os.environ, "EXEMPT_EMAIL_DOMAINS": "", "PYTHONDONTWRITEBYTECODE": "1"}


def sha(b):
    return hashlib.sha256(b).hexdigest()


def clear_pycache(path):
    shutil.rmtree(pathlib.Path(path).parent / "__pycache__", ignore_errors=True)


def run(labels, log_name):
    LOGS.mkdir(parents=True, exist_ok=True)
    log = LOGS / log_name
    with open(log, "wb") as fh:
        out = subprocess.run(
            [sys.executable, "manage.py", "test", *labels,
             "--settings=settings_worktree", "--noinput", "--keepdb", "-v1"],
            stdout=fh, stderr=subprocess.STDOUT, env=ENV, stdin=subprocess.DEVNULL,
        )
    o = log.read_text(errors="replace")
    blocks = {}
    for m in re.finditer(
        r"^={70}\n(?:FAIL|ERROR): (\w+) \(([\w.]+)\)[^\n]*\n(.*?)(?=^={70}\n|\Z)",
        o, re.M | re.S,
    ):
        method, dotted, body = m.groups()
        blocks[".".join(dotted.split(".")[-2:-1] + [method])] = body
    ran = next((l for l in o.splitlines() if l.startswith("Ran ")), "Ran ?")
    summary = next((l for l in o.splitlines() if l.startswith(("OK", "FAILED"))), f"? rc={out.returncode}")
    return out.returncode, ran, summary, blocks, o

    for name, (labels, mode, tests, frags) in EXPECTED.items():
        for t in tests:
            cls, meth = t.split(".")
            hits = 0
            for lab in labels:
                p = pathlib.Path(lab.replace(".", "/") + ".py")
                txt = p.read_text()
                if re.search(rf"^class {cls}\b", txt, re.M) and f"def {meth}(" in txt:
                    hits += 1
            print(" ", name, t, "found" if hits == 1 else f"HITS={hits}", flush=True)
    sys.exit()

    if counts != [1] * len(edits):
        print(name, "APPLY_FAILED", path, counts, flush=True)
        continue
    for old, new in edits:
        src = src.replace(old, new, 1)
    ast.parse(src)
    labels, mode, tests, frags = EXPECTED[name]
    try:
        clear_pycache(path)
        open(path, "w").write(src)
        rc, ran, summary, blocks, _ = run(labels, f"{name}.txt")
        failing = sorted(blocks)
        if rc == 0:
            status = "SURVIVED"
        elif ran == "Ran ?" or not failing:
            status = "BROKEN"
        elif mode == "EXACT":
            status = "KILLED" if failing == sorted(tests) else "KILLED_OTHER_SET"
        else:
            status = "KILLED" if all(t in failing for t in tests) else "KILLED_OTHER_SET"
        frag_bad = [t for t, rx in frags.items()
                    if t in blocks and not re.search(rx, blocks[t])]
        if status == "KILLED" and frag_bad:
            status = "FRAGMENT_MISSING"
        print(name, "|", status, "|", ran, "|", summary, "|", mode, tests,
              "| failing", failing, "| fragment-missing", frag_bad, flush=True)
    finally:
        subprocess.run(["git", "checkout", "--", path], check=True)
        clear_pycache(path)
        ok = sha(open(path, "rb").read()) == sha(
            subprocess.run(["git", "show", f"{COMMIT}:{path}"], capture_output=True).stdout
        )
        print("  restore", path, "ok" if ok else "MISMATCH", "| __pycache__ cleared", flush=True)
