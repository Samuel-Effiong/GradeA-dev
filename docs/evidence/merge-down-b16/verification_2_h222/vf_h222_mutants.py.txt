def sha(b):
    return hashlib.sha256(b).hexdigest()


def clear_pycache():
    for p in (SVC, LIC):
        shutil.rmtree(pathlib.Path(p).parent / "__pycache__", ignore_errors=True)


def run(labels, log_name):
    LOGS.mkdir(parents=True, exist_ok=True)
    log = LOGS / log_name
    with open(log, "wb") as fh:
        out = subprocess.run(
            [sys.executable, "manage.py", "test", *labels,
             "--settings=settings_worktree", "--noinput", "--keepdb", "-v1"],
            stdout=fh, stderr=subprocess.STDOUT, env=ENV, stdin=subprocess.DEVNULL)
    o = log.read_text(errors="replace")
    blocks = {}
    for m in re.finditer(
        r"^={70}\n(?:FAIL|ERROR): (\w+) \(([\w.]+)\)[^\n]*\n(.*?)(?=^={70}\n|\Z)",
        o, re.M | re.S):
        method, dotted, body = m.groups()
        blocks[".".join(dotted.split(".")[-2:-1] + [method])] = body
    ran = next((l for l in o.splitlines() if l.startswith("Ran ")), "Ran ?")
    summary = next((l for l in o.splitlines() if l.startswith(("OK", "FAILED"))), f"? rc={out.returncode}")
    return out.returncode, ran, summary, blocks, o
