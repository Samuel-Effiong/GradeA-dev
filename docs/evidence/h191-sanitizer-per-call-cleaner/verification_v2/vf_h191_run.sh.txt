#!/bin/bash
# v2's one slot for H-191 at the final tip (only after the Release Engineer's GRANT).
# Usage: vf_h191_run.sh <tip-sha8>   (start by full path)
set -uo pipefail
if [ -z "${GAP_INHIBITED:-}" ]; then GAP_INHIBITED=1 exec systemd-inhibit --what=idle:sleep:handle-lid-switch --who=GAP --why=GAP-test-run --mode=block "$0" "$@"; fi
TIP=$1
P=/home/bond-servant-in-training/Documents/Projects
WT=$P/Grade-Automator-Plus-vf2-s1
H=$P/GAP-v2-handover
PY=/home/bond-servant-in-training/Documents/Virtualenvs/AutoGrader_env/bin/python
PROBE=assignments/tests_vf2_h191_probe.py
OLDM=assignments/_vf2_old_pmc.py
export COMMIT=$TIP MUTANT_LOGS=$H/runs/h191_${TIP}_mutant_logs EXEMPT_EMAIL_DOMAINS= PYTHONDONTWRITEBYTECODE=1
ST=$H/runs/h191_${TIP}.status
WRAP="systemd-run --user --scope -q -p MemoryMax=6G -p MemorySwapMax=0 nice -n 10 timeout -k 60 1800"
say() { echo "$(date +%H:%M:%S) $*" >> $ST; }
cd $WT || exit 9
[ "$(git rev-parse --short=8 HEAD)" = "$TIP" ] && [ -z "$(git status --porcelain)" ] || { echo "NOT FROZEN at $TIP"; exit 9; }
[ ! -e $MUTANT_LOGS ] || { echo "logs of $TIP exist already"; exit 9; }
cp $H/tests_vf2_h191_probe.py $WT/$PROBE
git show 3f2ad13e:assignments/prosemirror_converter.py > $WT/$OLDM
cmp -s $H/tests_vf2_h191_probe.py $WT/$PROBE || { echo "probe copy differs"; exit 9; }
[ "$(git hash-object $WT/$OLDM)" = "$(git rev-parse 3f2ad13e:assignments/prosemirror_converter.py)" ] || { echo "old module differs"; exit 9; }
[ "$(git status --porcelain | sort)" = "$(printf '?? %s\n?? %s' $OLDM $PROBE | sort)" ] || { echo "unexpected files"; rm -f $WT/$PROBE $WT/$OLDM; exit 9; }
finish() {
  rm -f $WT/$PROBE $WT/$OLDM
  rm -rf $WT/assignments/__pycache__
  if [ -z "$(git -C $WT status --porcelain)" ] && [ "$(git -C $WT rev-parse --short=8 HEAD)" = "$TIP" ]; then say "probe removed; tree is the frozen tip again"; else say "WARNING: tree not the frozen tip at exit"; fi
}
trap finish EXIT
for step in check baseline mutants obs; do
  say "$step start: loadavg $(cat /proc/loadavg)"
  rc=0; $WRAP $PY -B $H/vf_h191_mutants.py $step > $H/runs/h191_${TIP}_$step.log 2>&1 < /dev/null || rc=$?
  say "$step end exit=$rc: loadavg $(cat /proc/loadavg)"
  [ $step = baseline ] && [ $rc -ne 0 ] && { echo "baseline red: nothing more is run"; exit 1; }
done
echo "done: see $ST"
