"""Mutation battery for the teacher-detail credit conversion fix
(dashboard/views.py: SchoolAdminDashboardView.teacher_detail). Each
mutant reverts ONE // CONVERSION_FACTOR division; a test asserting that
field's divided value must fail. Restores from a collision-safe backup
copy and verifies by md5 (never git checkout).
"""
import hashlib, json, os, re, shutil, subprocess, sys, time

ROOT = "/home/bond-servant-in-training/Documents/Projects/GAP-fix-teacher-credits-conversion"
SCR = os.path.dirname(os.path.abspath(__file__))
DV = "dashboard/views.py"
LABELS = "dashboard.tests.TeacherDetailAPITest"

MUTANTS = [
    ("CAT", "category amount not converted",
     [(DV, '"amount": amount // CONVERSION_FACTOR,', '"amount": amount,')],
     "test_category_amount_is_the_raw_total_divided_by_conversion_factor"),
    ("USED", "credits_used not converted",
     [(DV, 'result["credits_used"] = credits_used // CONVERSION_FACTOR',
           'result["credits_used"] = credits_used')],
     "test_credits_used_is_the_raw_total_divided_by_conversion_factor"),
    ("MONTH", "credits_remaining.monthly not converted",
     [(DV, '"monthly": credits_remaining_monthly // CONVERSION_FACTOR,',
           '"monthly": credits_remaining_monthly,')],
     "test_credits_remaining_is_the_raw_total_divided_by_conversion_factor"),
    ("CARRY", "credits_remaining.carry_over not converted",
     [(DV, '"carry_over": credits_remaining_carry_over // CONVERSION_FACTOR,',
           '"carry_over": credits_remaining_carry_over,')],
     "test_credits_remaining_is_the_raw_total_divided_by_conversion_factor"),
    ("OVER", "credits_remaining.overage not converted",
     [(DV, '"overage": credits_remaining_overage // CONVERSION_FACTOR,',
           '"overage": credits_remaining_overage,')],
     "test_credits_remaining_is_the_raw_total_divided_by_conversion_factor"),
    ("TOTAL", "credits_remaining.total floors the sum of PARTS instead of the raw total",
     [(DV, '            )\n            // CONVERSION_FACTOR,\n        }',
           '            ),\n        }')],
     "test_credits_remaining_is_the_raw_total_divided_by_conversion_factor"),
    ("DAILY", "daily_usage credits not converted",
     [(DV, 'usage_dict.get(window_start + timedelta(days=i), 0)\n                // CONVERSION_FACTOR,',
           'usage_dict.get(window_start + timedelta(days=i), 0),')],
     "test_daily_usage_credits_is_the_raw_total_divided_by_conversion_factor"),
]


def md5(p):
    return hashlib.md5(open(p, "rb").read()).hexdigest()


def run(timeout=300):
    env = dict(os.environ)
    env["DJANGO_SETTINGS_MODULE"] = "settings_worktree"
    r = subprocess.run(
        [sys.executable, "manage.py", "test", *LABELS.split(),
         "--settings=settings_worktree", "--keepdb", "--noinput"],
        cwd=ROOT, capture_output=True, text=True, timeout=timeout, env=env,
    )
    out = r.stdout + r.stderr
    fails = re.findall(r"^(?:FAIL|ERROR): (\S+) \(([^)]*)\)", out, re.M)
    m = re.search(r"^(OK.*|FAILED.*)$", out, re.M)
    return r.returncode, out, fails, (m.group(1) if m else "?")


def precheck():
    bad = []
    for mid, desc, edits, expect in MUTANTS:
        for f, old, new in edits:
            n = open(os.path.join(ROOT, f)).read().count(old)
            if n != 1:
                bad.append((mid, f, n, old[:60]))
    return bad


def main(only=None):
    bad = precheck()
    if bad:
        for b in bad:
            print("PATTERN PROBLEM", b)
        raise SystemExit(1)
    log = open(os.path.join(SCR, "mutation_log.jsonl"), "a")
    results = []
    files = sorted({f for _, _, ms, _ in MUTANTS for f, _, _ in ms})
    orig = {f: md5(os.path.join(ROOT, f)) for f in files}
    backups = {}
    for f in files:
        b = os.path.join(SCR, "backup_" + f.replace("/", "__"))
        shutil.copy2(os.path.join(ROOT, f), b)
        backups[f] = b
    rc, out, fails, summary = run()
    base = {"id": "BASE", "rc": rc, "summary": summary, "fails": fails}
    print("BASELINE", rc, summary, flush=True)
    log.write(json.dumps(base) + "\n")
    if rc != 0:
        print(out[-3000:])
        raise SystemExit("baseline not green")
    for mid, desc, edits, expect in MUTANTS:
        if only and mid not in only:
            continue
        try:
            for f, old, new in edits:
                p = os.path.join(ROOT, f)
                s = open(p).read()
                open(p, "w").write(s.replace(old, new))
            t = time.time()
            rc, out, fails, summary = run()
            names = sorted({n for n, _ in fails})
            killed = rc != 0 and any(expect in n for n, _ in fails)
            res = {"id": mid, "desc": desc, "rc": rc, "summary": summary, "expected_test": expect,
                   "failing": names, "killed_for_named_reason": killed, "secs": round(time.time() - t)}
        except Exception as e:
            res = {"id": mid, "desc": desc, "error": str(e), "killed_for_named_reason": False}
        finally:
            for f in files:
                shutil.copy2(backups[f], os.path.join(ROOT, f))
            bad = [f for f in files if md5(os.path.join(ROOT, f)) != orig[f]]
            if bad:
                raise SystemExit(f"RESTORE MISMATCH after {mid}: {bad}")
        res["restored_md5_ok"] = True
        print(mid, "KILLED" if res["killed_for_named_reason"] else "SURVIVED/ERR",
              (res.get("failing") or res.get("error"))[:6] if res.get("failing") else res.get("error"), flush=True)
        log.write(json.dumps(res) + "\n")
        log.flush()
        results.append(res)
    json.dump({"baseline": base, "mutants": results, "md5": orig},
               open(os.path.join(SCR, "mutation_results.json"), "w"), indent=1)
    print("KILLED", sum(r["killed_for_named_reason"] for r in results), "/", len(results))


if __name__ == "__main__":
    main(set(sys.argv[1:]) or None)
