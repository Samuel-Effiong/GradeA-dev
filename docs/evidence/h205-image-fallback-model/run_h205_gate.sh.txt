#!/bin/bash
# H-205 image-capable backup (beta line). Usage: bash run_h205_gate.sh <tip> 1
# One grant, inside ONE outer inhibit (memory cap = available - 2G, min 6G, read at start; rule 13):
#   systemd-inhibit --what=idle:sleep:handle-lid-switch --who=GAP --why=GAP-test-run --mode=block bash <script> <tip> 1
# 0 the red run: the five test modules as at the tip against the code as at the base (the nine production files
#   taken back; own DB). RULE 22: each new red test is red for its WRITTEN reason (a fragment looked for in its
#   own FAIL/ERROR block);
# 1a makemigrations --check; 1b the seam by program (the diff of ai_processor/services.py removes exactly the old
#   six-line selection and its comment line; nothing else is removed); 1 the five modules + near modules (every
#   module that reads the fallback lists, the configs, the grading label or the second opinion) + guards;
#   2 the 14 mutants (rule 17; own DB).
# The regression is the batch's one full run on the frozen tip; it is not in this script.
# A trap on EXIT restores the tree. RULE 18: every run writes straight to a file, stdin from /dev/null, nothing piped.
set -u
# BASE = origin/beta after batch 22; the red run takes the nine production files back to it.
TIP=$1; STEP=${2:-1}; BASE=dac16545
cd /home/bond-servant-in-training/Documents/Projects/Grade-Automator-Plus-image-fallback-model || exit 1
source /home/bond-servant-in-training/Documents/Virtualenvs/AutoGrader_env/bin/activate
E=docs/evidence/h205-image-fallback-model
AVAIL_G=$(awk '/^MemAvailable:/ {print int($2/1048576)}' /proc/meminfo)
MEMG=$((AVAIL_G - 2))
echo "available at start: ${AVAIL_G}G; cap ${MEMG}G"
[ "$MEMG" -ge 6 ] || { echo "STOP (exit 5): cap under 6G; wait and tell the SM"; exit 5; }
# Each run has its own scope unit, gap-h205-<pid>-<part>; the watchdogs stop only those, by exact name.
UNITP=gap-h205-$$
UPART=none
CAP="systemd-run --user --scope -q --unit=\${UNITP}-\${UPART} -p MemoryMax=${MEMG}G -p MemorySwapMax=0"
# Memory watchdog (SM 2026-10-14): available under 1.5 GB stops OUR scope, by its unit name, never by pattern.
( while sleep 10; do
    A=$(awk '/^MemAvailable:/ {print $2}' /proc/meminfo)
    if [ "$A" -lt 1572864 ]; then
        for U in $UNITP-red $UNITP-modules $UNITP-mutants; do
            if systemctl --user is-active -q "$U.scope"; then
                echo "MEMORY STOP $(date +%T): available ${A} kB; stopping $U.scope" >> $E/memory_watchdog.txt
                systemctl --user stop "$U.scope"
            fi
        done
    fi
  done ) &
WATCHDOG=$!
export PYTHONFAULTHANDLER=1
export EXEMPT_EMAIL_DOMAINS=
[ "$(git rev-parse HEAD)" = "$(git rev-parse $TIP)" ] || { echo "not at $TIP"; exit 1; }
git diff --quiet HEAD || { echo "tree not clean"; exit 1; }
[ "$STEP" = 1 ] || { echo "only step 1 exists"; exit 1; }
mkdir -p $E
CHANGED="ai_processor/services.py ai_processor/grading_run.py audit/emitter.py ai_processor/grading_config.py ai_processor/reading_config.py AutoGrader/settings.py students/services.py ai_processor/benchmark/answers/live.py ai_processor/grading_cache.py"
restore_tree() { kill $WATCHDOG 2>/dev/null; git checkout HEAD -- $CHANGED 2>/dev/null; find ai_processor -name __pycache__ -type d -prune -exec rm -rf {} + 2>/dev/null; }
trap restore_tree EXIT
summary() { grep -E "^(Ran |OK|FAILED \(|(FAIL|ERROR): )" "$1" | cut -c1-200 | head -${2:-12}; }
dropmut() { python manage.py shell --settings=settings_worktree_mut -c "
from django.conf import settings
from django.db import connection
name = settings.DATABASES['default']['TEST']['NAME']
with connection._nodb_cursor() as c:
    c.execute('DROP DATABASE IF EXISTS \"%s\"' % name)
print('dropped', name)" < /dev/null > $E/.dropmut.out 2>&1; tail -1 $E/.dropmut.out; rm -f $E/.dropmut.out; }
OWN="ai_processor.tests_h205_image_fallback ai_processor.tests_grading_config ai_processor.tests_reading_config ai_processor.tests_grading_hardening audit.tests_metrics"
NEAR="billing.tests.test_execute_graded_task ai_processor.tests_grading_run_label ai_processor.tests_grading_run_pipeline ai_processor.tests_grading_run_checker ai_processor.tests_vote_helper ai_processor.tests_second_opinion_pipeline students.tests_second_opinion_queue students.tests_grading_label_end_to_end students.tests_grading_label_routes students.tests_grading_label_written ai_processor.tests_grading_cache ai_processor.tests_grading_cache_key_v2 ai_processor.tests_call_record ai_processor.tests_call_record_static ai_processor.tests_provider_block ai_processor.tests_ai_call_trace ai_processor.tests_reading_pipeline ai_processor.tests_reading_run ai_processor.tests_h242_retranscription ai_processor.tests_h253_answers_never_empty students.tests_reading_label_static"
GUARDS="AutoGrader.tests_cache_bespoke_1114 AutoGrader.tests_no_wildcard_invalidation AutoGrader.tests_cache_invalidation_coverage AutoGrader.tests_migration_rollback_defaults AutoGrader.tests_migration_safety_check AutoGrader.tests_redis_test_isolation AutoGrader.tests_beat_health AutoGrader.tests_beat_locks AutoGrader.tests_management_commands_are_commands AutoGrader.tests_error_messages AutoGrader.tests_no_pii_in_logs AutoGrader.tests_log_scrubbing AutoGrader.tests_sentry_scrubbing AutoGrader.tests_redis_hygiene_databases AutoGrader.tests_redis_hygiene AutoGrader.tests_patient_test_stream AutoGrader.tests_pool_worker_sigterm AutoGrader.tests_student_feedback_guard AutoGrader.tests_submission_audience_guard AutoGrader.tests_student_classmates_guard AutoGrader.tests_network_guard"
echo "load at start: $(cat /proc/loadavg)"
echo "=== 0. the red run: $OWN against the code as at $BASE (own DB) $(date +%T)"
git checkout $BASE -- $CHANGED
find ai_processor -name __pycache__ -type d -prune -exec rm -rf {} +
UPART=red
eval "PYTHONDONTWRITEBYTECODE=1 $CAP nice -n 10 timeout -k 60 1800 python manage.py test --noinput --verbosity 2 --settings=settings_worktree_mut $OWN" < /dev/null > $E/red_run_code_as_at_base.txt 2>&1
RED_EXIT=$?
echo "red run exit=$RED_EXIT (non-zero expected)"
RF=$E/red_run_code_as_at_base.txt
RC=/home/bond-servant-in-training/Documents/Projects/GAP-builder-scripts/reason_check.py
RED_OK=1
[ "$RED_EXIT" -ne 0 ] || RED_OK=0
grep -qE "^OK" $RF && RED_OK=0
grep -qE "^Ran [0-9]+ tests? in" $RF || { echo "red run: no Ran line"; RED_OK=0; }
grep -qE "unittest.loader._FailedTest" $RF && { echo "red run: the module failed to load"; RED_OK=0; }
# RULE 22: each test below is red for its WRITTEN reason (a fragment written beforehand, looked for in its own block).
# Tests that PASS at the base by design (guards that hold on the code as it is): the_rest_of_the_request_is_unchanged,
# the_override_and_the_fallback_list_do_not_change_the_block, the_only_create_call_is_the_one_in_ai_model,
# every_file_..._is_on_the_named_list, every_listed_file_..._carries_a_mark_and_a_reason,
# every_listed_file_other_than_services_has_zero_create_calls, the_provider_host_appears_only_in_listed_files
r() { python -I $RC $RF "$1" "$2" || RED_OK=0; }
r test_an_image_in_the_user_prompt_of_each_page_task "['deepseek/deepseek-v4-pro'] != ['anthropic/claude-haiku-5.5']"
r test_an_image_only_in_a_message "['deepseek/deepseek-v4-pro'] != ['anthropic/claude-haiku-5.5']"
r test_an_image_only_in_a_list_system_prompt "['deepseek/deepseek-v4-pro'] != ['anthropic/claude-haiku-5.5']"
r test_an_image_call_sends_haiku_as_the_only_backup "['deepseek/deepseek-v4-pro'] != ['anthropic/claude-haiku-5.5']"
r test_the_image_backup_is_haiku_and_never_a_nano_model "has no attribute 'IMAGE_FALLBACK_MODELS'"
r test_a_grading_answered_by_haiku_says_a_backup_was_used "'unknown' != 'yes'"
r test_the_grading_version_changes_with_the_image_list "does not have the attribute 'IMAGE_FALLBACK_MODELS'"
r test_the_reading_version_changes_with_the_image_list "does not have the attribute 'IMAGE_FALLBACK_MODELS'"
r test_the_default_candidates_are_deepseek_then_haiku "'google/gemini-2.5-flash'"
r test_no_production_file_names_gemini_2_5_flash "'AutoGrader/settings.py'"
r test_the_other_values_cover_every_name "'IMAGE_FALLBACK_MODELS'"
r test_the_version_for_a_fixed_reading_is_pinned "does not have the attribute 'IMAGE_FALLBACK_MODELS'"
r test_each_constant_that_shapes_a_reading_changes_the_version "does not have the attribute 'IMAGE_FALLBACK_MODELS'"
r test_image_fallbacks_never_include_a_nano_model "has no attribute 'IMAGE_FALLBACK_MODELS'"
r test_a_completed_grading_on_the_image_backup_emits_fallback_rate_one "call('model_fallback_rate', 0.0)"
if [ "$RED_OK" != 1 ]; then
    echo "STOP (exit 4): the red run is not what was written"
    git checkout HEAD -- $CHANGED
    find ai_processor -name __pycache__ -type d -prune -exec rm -rf {} +
    exit 4
fi
echo "red run is as written: non-zero exit, no OK line, the module loaded, fifteen tests red for their written reasons"
git checkout HEAD -- $CHANGED
find ai_processor -name __pycache__ -type d -prune -exec rm -rf {} +
git diff --quiet HEAD -- . ":!$E" && [ -z "$(git status --porcelain -- . ":!$E" | grep -v settings_worktree_mut.py)" ] && echo "source restored" || { echo "STOP: not restored"; exit 3; }
grep -E "^(Ran |OK|FAILED \\()" $RF | cut -c1-200
dropmut
echo "=== 1a. makemigrations --check $(date +%T)"
python manage.py makemigrations --check --dry-run --settings=settings_worktree < /dev/null > $E/makemigrations_check.txt 2>&1; MM=$?; echo "makemigrations exit=$MM (0 = no changes)"; tail -1 $E/makemigrations_check.txt
[ "$MM" = 0 ] || { echo "STOP: makemigrations reports changes"; exit 2; }
echo "=== 1b. the seam: what the diff of services.py removes, by program"
git diff $BASE HEAD -U0 -- ai_processor/services.py | grep -E '^-[^-]' > $E/seam_removed_lines.txt
cat $E/seam_removed_lines.txt
[ "$(wc -l < $E/seam_removed_lines.txt)" = 6 ] || { echo "STOP: the diff removes other than the old selection (5 lines) and its comment line (1)"; exit 2; }
[ "$(grep -c 'else None' $E/seam_removed_lines.txt)" = 1 ] || { echo "STOP: the old selection is not what is removed"; exit 2; }
echo "=== 1. own modules + near modules + guards at $(git rev-parse --short=8 HEAD) $(date +%T)"
UPART=modules
eval "$CAP nice -n 10 timeout -k 60 3500 python manage.py test --noinput --verbosity 2 --settings=settings_worktree $OWN $NEAR $GUARDS" < /dev/null > $E/modules_and_guards.txt 2>&1
echo "exit=$? $(date +%T)"; summary $E/modules_and_guards.txt
if grep -qE "^FAILED \\(" $E/modules_and_guards.txt || ! grep -qE "^OK" $E/modules_and_guards.txt; then echo "STOP: not green"; exit 2; fi
echo "=== 2. mutants: 14 (own DB, rules 17 and 18) $(date +%T)"
# Time limit 90 s x 14 + 600 = 1860 s, with a 300 s stall watchdog on the newest mutant log (2026-10-14).
UPART=mutants
( sleep 60; while systemctl --user is-active -q "$UNITP-mutants.scope"; do
    NEWEST=$(ls -t $E/mutant_logs/*.txt 2>/dev/null | head -1)
    if [ -n "$NEWEST" ] && [ $(( $(date +%s) - $(stat -c %Y "$NEWEST") )) -gt 300 ]; then
        echo "STALL STOP $(date +%T): $NEWEST silent over 300 s; stopping $UNITP-mutants.scope" >> $E/stall_watchdog.txt
        systemctl --user stop "$UNITP-mutants.scope"; break
    fi
    sleep 15
  done ) &
STALL=$!
eval "MUT_SETTINGS=settings_worktree_mut MUT_RESULTS=$E/mutation_results.json MUT_LOGS=$E/mutant_logs $CAP nice -n 10 timeout -k 60 1860 python $E/mutate.py" < /dev/null > $E/mutation_log.txt 2>&1
echo "mutants exit=$? $(date +%T)"; kill $STALL 2>/dev/null; cut -c1-260 $E/mutation_log.txt
git diff --quiet HEAD -- . ":!$E" && echo "source clean after mutants" || echo "STOP: source not clean after mutants"
dropmut
echo "load at end: $(cat /proc/loadavg)"
