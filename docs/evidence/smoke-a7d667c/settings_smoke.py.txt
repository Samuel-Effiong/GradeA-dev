Everything real except the three things that would leave the machine or cost

    "default": {"BACKEND": "django.core.files.storage.FileSystemStorage"},

# Route every OpenAI-SDK client (ai_processor builds one in AIProcessor.__init__)
# to the local fake provider, whatever base_url the code asks for.
import openai  # noqa: E402

_RealOpenAI = openai.OpenAI


class _FakeRoutedOpenAI(_RealOpenAI):
    def __init__(self, *args, **kwargs):
        kwargs["base_url"] = os.environ["SMOKE_FAKE_PROVIDER_URL"]
        kwargs["api_key"] = "sk-smoke-dummy"
        super().__init__(*args, **kwargs)


openai.OpenAI = _FakeRoutedOpenAI

# tiktoken cannot download cl100k_base here (SSL EOF). Only .encode() is used
# by the app; an approximate stand-in keeps the estimate path running.
import tiktoken  # noqa: E402


class _ApproxEncoding:
    def encode(self, text, **kwargs):
        return list(range(max(1, len(text) // 4)))


tiktoken.get_encoding = lambda name: _ApproxEncoding()
