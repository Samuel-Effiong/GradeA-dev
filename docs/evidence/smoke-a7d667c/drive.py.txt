def login(email, pw=PW):
    return requests.post(f"{B}/auth/login", json={"email": email, "password": pw})
_TOK = {}
def tok(email):
    if email not in _TOK:
        for _ in range(8):
            r = login(email)
            if r.status_code != 429: break
            time.sleep(5)   # the app's own login throttle; wait it out
        assert r.status_code == 200, (email, r.status_code, r.text[:200])
        _TOK[email] = {"Authorization": "Bearer " + r.json()["data"]["access"]}
    return _TOK[email]
def body(r):
    try: j = r.json()
    except Exception: return r.text[:300]
    return j
