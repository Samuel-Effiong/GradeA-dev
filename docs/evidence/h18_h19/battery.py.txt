def porcelain(cwd):
    """git status, ignoring ONLY the shared team/ symlink other sessions add."""
    lines = git(cwd, "status", "--porcelain").splitlines()
    return "\n".join(l for l in lines if l.strip() != "?? team")


def git(cwd, *args, **kw):
    return subprocess.run(
        ["git", *args], cwd=cwd, capture_output=True, text=True, check=True, **kw
    ).stdout


def blob_of_commit(path, rev=COMMIT):
    return git(MAIN, "rev-parse", f"{rev}:{path}").strip()


def blob_of_file(worker, path):
    return git(worker, "hash-object", path).strip()


def expand(name):
    spec = MUTANTS[name][2]
    if spec == "ORIGINAL":
        return "ORIGINAL"
    if spec == "REMOVE_TYPE_ONLY_BRANCH":
        return "REMOVE_TYPE_ONLY_BRANCH"
    if isinstance(spec, str):
        edits = []
        for part in spec.split("+"):
            key = next(k for k in MUTANTS if k.split(" ")[0] == part)
            edits += expand(key)
        return edits
    return spec


def apply(worker, name):
    spec = expand(name)
    touched = set()
    if spec == "ORIGINAL":
        for f in SOURCES:
            with open(os.path.join(worker, f), "w") as fh:
                fh.write(git(MAIN, "show", f"{PRE_FIX}:{f}"))
            touched.add(f)
        return touched
    if spec == "REMOVE_TYPE_ONLY_BRANCH":
        f = "ai_processor/services.py"
        p = os.path.join(worker, f)
        src = open(p).read()
        start = src.index(TYPE_ONLY_BRANCH_HEAD)
        end = src.index(
            "        elif user.user_type == UserTypes.SUPER_ADMIN:\n", start
        )
        assert src.count(TYPE_ONLY_BRANCH_HEAD) == 1
        open(p, "w").write(src[:start] + src[end:])
        return {f}
    for f, old, new in spec:
        p = os.path.join(worker, f)
        src = open(p).read()
        n = src.count(old)
        assert n == 1, f"{name}: anchor in {f} occurs {n}x"
        open(p, "w").write(src.replace(old, new))
        touched.add(f)
    return touched


def restore(worker, files):
    verdicts = {}
    for f in files:
        with open(os.path.join(worker, f), "w") as fh:
            fh.write(git(MAIN, "show", f"{COMMIT}:{f}"))
        verdicts[f] = blob_of_file(worker, f) == blob_of_commit(f)
    return verdicts


def make_worker(i):
    path = f"{PARENT}/Grade-Automator-Plus-h18mut-w{i}"
    if not os.path.isdir(path):
        git(MAIN, "worktree", "add", "--detach", path, COMMIT)
        os.symlink(f"{MAIN}/.env", f"{path}/.env")
        with open(f"{path}/settings_worktree.py", "w") as fh:
            fh.write(
                "from AutoGrader.settings import *  # noqa: F401,F403\n"
                "from AutoGrader.settings import DATABASES\n\n"
                'DATABASES["default"].setdefault("TEST", {})\n'
                f'DATABASES["default"]["TEST"]["NAME"] = "test_h18mut_w{i}"\n'
            )
    assert git(path, "rev-parse", "HEAD").strip() == COMMIT
    assert porcelain(path) == "", f"worker {i} dirty"
    return path


def run_tests(worker, tests, log):
    r = subprocess.run(
        [
            PY,
            "manage.py",
            "test",
            *tests,
            "--settings=settings_worktree",
            "--noinput",
            "--keepdb",
        ],
        cwd=worker,
        capture_output=True,
        text=True,
    )
    out = r.stdout + r.stderr
    with open(log, "w") as fh:
        fh.write(out)
    ran = re.search(r"^Ran (\d+) tests", out, re.M)
    summ = re.findall(r"^(OK.*|FAILED.*)$", out, re.M)
    failing = sorted(set(re.findall(r"^(?:FAIL|ERROR): (\S+) \(([^)]+)\)", out, re.M)))
    return (
        r.returncode,
        (ran.group(1) if ran else "?"),
        (summ[-1] if summ else "?"),
        failing,
    )


def main():
    names = [n for n in MUTANTS if MUTANTS[n][0] is not None]
    if len(sys.argv) > 1:
        names = [n for n in names if any(n.startswith(p) for p in sys.argv[1:])]
    workers = [make_worker(i) for i in range(K)]
    work = queue.Queue()
    for n in names:
        work.put(n)
    results = {}
    lock = threading.Lock()

    def loop(worker, idx):
        # Warm the worker DB once (migration cost) with a clean run of one module.
        rc, ran, summ, _ = run_tests(worker, FAILCLOSED, f"{OUT}/w{idx}_warm.log")
        with lock:
            print(f"[w{idx}] warm clean run: rc={rc} ran={ran} {summ}", flush=True)
        while True:
            try:
                name = work.get_nowait()
            except queue.Empty:
                return
            guard, tests, _ = MUTANTS[name]
            touched = set()
            t0 = time.time()
            try:
                touched = apply(worker, name)
                rc, ran, summ, failing = run_tests(
                    worker, tests, f"{OUT}/{name.split(' ')[0]}.log"
                )
            finally:
                verdicts = restore(worker, touched) if touched else {}
            clean = porcelain(worker) == ""
            verdict = "KILLED" if rc != 0 else "SURVIVED"
            with lock:
                results[name] = dict(
                    guard=guard,
                    verdict=verdict,
                    ran=ran,
                    summary=summ,
                    failing=[f"{c.rsplit('.', 1)[-1]}.{t}" for t, c in failing],
                    restored=all(verdicts.values()) and clean,
                    restore_detail=verdicts,
                    worker=idx,
                    seconds=round(time.time() - t0),
                )
                print(
                    f"{verdict:8} {name} | guard={guard} ran={ran} {summ} restored={results[name]['restored']} ({results[name]['seconds']}s)",
                    flush=True,
                )

    threads = [
        threading.Thread(target=loop, args=(w, i)) for i, w in enumerate(workers)
    ]
    for t in threads:
        t.start()
    for t in threads:
        t.join()

    with open(f"{OUT}/results.json", "w") as fh:
        json.dump(results, fh, indent=1, sort_keys=True)
    killed = sum(r["verdict"] == "KILLED" for r in results.values())
    restored = all(r["restored"] for r in results.values())
    print(
        f"\nkilled {killed}/{len(results)}; all restored by blob hash: {restored}",
        flush=True,
    )
    for i, w in enumerate(workers):
        final_clean = porcelain(w) == ""
        head = git(w, "rev-parse", "HEAD").strip()
        print(f"worker {i}: HEAD={head} clean={final_clean}", flush=True)


if __name__ == "__main__":
    main()
