#!/bin/bash
# H-209 small slot before the fourth chain run (SM ruling): only the two test
# modules whose D8/X1 tests were amended, to see them green. Rule 18: output
# straight to a file, stdin from /dev/null. Usage: small.sh <tip-sha8>
set -euo pipefail
TIP=$1
P=/home/bond-servant-in-training/Documents/Projects
WT=$P/Grade-Automator-Plus-h208-broker-text-not-logged-h209-task-results-no-text
OUT=$P/GAP-d5-runs/h209
export EXEMPT_EMAIL_DOMAINS=
WRAP="systemd-inhibit --what=idle:sleep:handle-lid-switch --who=GAP --why=GAP-test-run --mode=block systemd-run --user --scope -q -p MemoryMax=6G -p MemorySwapMax=0 nice -n 10 timeout -k 60 1800"
say() { echo "$(date +%H:%M:%S) $*" | tee -a $OUT/small.status; }
cd $WT
[ "$(git rev-parse --short=8 HEAD)" = "$TIP" ] && [ -z "$(git status --porcelain)" ] || { say "ABORT: tree not frozen at $TIP"; exit 9; }
[ ! -e $OUT/small_$TIP.log ] || { say "ABORT: small_$TIP.log exists already"; exit 9; }
say "small start on $TIP: loadavg $(cat /proc/loadavg)"
rc=0; s=$(date +%s)
PYTHONDONTWRITEBYTECODE=1 $WRAP python manage.py test assignments.tests_task_results_carry_no_text assignments.tests_course_ownership_failure students.tests_h38_tasks_namespace --settings=settings_worktree --noinput > $OUT/small_$TIP.log 2>&1 < /dev/null || rc=$?
echo "exit=$rc wall_s=$(( $(date +%s)-s ))" >> $OUT/small_$TIP.log
say "small end: exit=$rc loadavg $(cat /proc/loadavg)"
say "$(grep -E '^(Ran |OK|FAILED)' $OUT/small_$TIP.log | tr '\n' ' ')"
say "SMALL END"
exit $rc
