ENV = {**os.environ, "EXEMPT_EMAIL_DOMAINS": "", "PYTHONDONTWRITEBYTECODE": "1"}
sha = lambda b: hashlib.sha256(b).hexdigest()
def clear_pycache(path): shutil.rmtree(pathlib.Path(path).parent / "__pycache__", ignore_errors=True)

def run(log_name, labels=None):
    LOGS.mkdir(parents=True, exist_ok=True)
    log = LOGS / log_name
    with open(log, "wb") as fh:
        out = subprocess.run([sys.executable, "manage.py", "test", *(labels or LABEL), "--settings=settings_worktree", "--noinput", "--keepdb", "-v1"],
                             stdout=fh, stderr=subprocess.STDOUT, env=ENV, stdin=subprocess.DEVNULL)
    o = log.read_text(errors="replace")
    failing = set(re.findall(r"^(?:FAIL|ERROR): (\w+)", o, re.M))
    ran = next((l for l in o.splitlines() if l.startswith("Ran ")), "Ran ?")
    summary = next((l for l in o.splitlines() if l.startswith(("OK", "FAILED"))), f"? rc={out.returncode}")
    return out.returncode, ran, summary, failing, o

def frag_seen(o, test, rx):
    for blk in re.split(r"^={50,}$", o, flags=re.M):
        if re.search(rf"^(?:FAIL|ERROR): {test} ", blk, re.M) and re.search(rx, blk):
            return True
    return False

def apply(src, edits):
    counts = [src.count(old) for old, _ in edits]
    if counts != [1] * len(edits):
        return None, counts
    for old, new in edits:
        src = src.replace(old, new, 1)
    ast.parse(src)
    return src, counts

step = sys.argv[1]
if step == "check":
    for name, (edits, _, _) in MUTANTS.items():
        s, c = apply(open(T).read(), edits)
        print(name, c, "APPLIES" if s else "NO_APPLY", "changed" if s and s != open(T).read() else "UNCHANGED")
    sys.exit(0)
if step == "baseline":
    clear_pycache(T)
    rc, ran, summary, failing, o = run("baseline.txt")
    print("baseline |", ran, "|", summary, "|", sorted(failing), flush=True)
    if rc != 0: print(o[-8000:])
    sys.exit(rc)
if step == "findings":
    rc, ran, summary, failing, o = run("findings.txt", FLABEL)
    print("findings |", ran, "|", summary, "|", sorted(failing), "| fragment seen", frag_seen(o, "test_f1_celerys_own_failure_log_for_a_failed_grade_all_carries_no_text", r"the made-up error text reached celery's own failure log"), flush=True)
    sys.exit(0)
for name, (edits, expected, frags) in MUTANTS.items():
    src, counts = apply(open(T).read(), edits)
    if src is None:
        print(name, "APPLY_FAILED", counts, flush=True); continue
    try:
        clear_pycache(T); open(T, "w").write(src)
        rc, ran, summary, failing, o = run(f"{name}.txt")
        seen = {t: frag_seen(o, t, rx) for t, rx in frags.items()}
        if rc == 0: status = "SURVIVED"
        elif ran == "Ran ?" or not failing: status = "BROKEN"
        elif failing == expected and all(seen.values()): status = "KILLED"
        else: status = "KILLED_OTHER"
        print(name, "|", status, "|", ran, "|", summary, "| expected", sorted(expected), "| failing", sorted(failing), "| fragments seen", seen, flush=True)
    finally:
        subprocess.run(["git", "checkout", "--", T], check=True); clear_pycache(T)
        ok = sha(open(T, "rb").read()) == sha(subprocess.run(["git", "show", f"{COMMIT}:{T}"], capture_output=True).stdout)
        print("  restore", T, "ok" if ok else "MISMATCH", flush=True)
