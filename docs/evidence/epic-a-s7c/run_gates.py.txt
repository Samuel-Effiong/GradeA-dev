"""S7c gates (author 1a): repro | modules | mutants | regression."""
import hashlib, os, pathlib, subprocess, sys, time

W = pathlib.Path("/home/bond-servant-in-training/Documents/Projects/Grade-Automator-Plus-epic-a-s7c")
PY = "/home/bond-servant-in-training/Documents/Virtualenvs/AutoGrader_env/bin/python"
BASE = "ca35971"  # epic tip before S7c (contains S7a, S7b)
PROD = ["AutoGrader/error_messages.py", "assignments/tasks.py", "students/exceptions.py",
        "students/item_retry.py", "students/models.py", "students/task_tracking.py"]
NEW = ["students.tests_credits_mid_batch"]
MODULES = list(dict.fromkeys(NEW + [
    "students.tests_item_retry", "students.tests_item_retry_h38", "students.tests_batch_item_results",
    "AutoGrader.tests_reason_codes", "AutoGrader.tests_codederror_serialization",
    "AutoGrader.tests_error_messages",
    "AutoGrader.tests_no_wildcard_invalidation", "AutoGrader.tests_cache_invalidation_coverage",
    "AutoGrader.tests_migration_rollback_defaults", "AutoGrader.tests_redis_test_isolation",
    "audit.tests_route_coverage", "audit.tests_history_guard",
    "classrooms.tests_teacher_access_sweep", "classrooms.tests_course_roster_scope_sweep",
]))
ns = {}
exec(open(pathlib.Path(__file__).with_name("s7c_mutants.py")).read(), ns)
MUTANTS = ns["MUTANTS"]

def test(labels, tag, settings="settings_worktree"):
    cmd = ["nice", "-n", "10", "timeout", "-k", "60", "1800", PY, "manage.py", "test",
           f"--settings={settings}", "--noinput", "-v", "2", *labels]
    # The mutants' settings module lives in the scratchpad, outside the worktree.
    env = {**os.environ, "PYTHONPATH": str(pathlib.Path(__file__).resolve().parent)}
    p = subprocess.run(cmd, cwd=W, capture_output=True, text=True, env=env)
    out = p.stdout + p.stderr
    (LOGS / f"{tag}.log").write_text(out)
    keep = [l for l in out.splitlines() if l.startswith(("Ran ", "OK", "FAILED", "FAIL:", "ERROR:"))]
    print(f"=== {tag}: exit {p.returncode}\n" + "\n".join(keep), flush=True)
    return p.returncode

def sha():
    return {f: hashlib.sha256((W / f).read_bytes()).hexdigest() for f in PROD}

LOGS = pathlib.Path(sys.argv[2]); LOGS.mkdir(parents=True, exist_ok=True)
mode = sys.argv[1]
# regression: which apps, and a tag for the log names (argv 3 and 4).
APPS = sys.argv[3].split(",") if len(sys.argv) > 3 else ["students", "users", "assignments"]
TAG = sys.argv[4] if len(sys.argv) > 4 else "5bef042"
before = sha()
try:
    if mode == "repro":
        # S7c's production code reverted to the base (the migration stays, so
        # the test DB still has the column; the model doesn't know it).
        for f in PROD:
            old = subprocess.run(["git", "show", f"{BASE}:{f}"], cwd=W, capture_output=True, text=True, check=True).stdout
            (W / f).write_text(old)
        test(NEW, "repro_s7c_reverted")
    elif mode == "repro_fix":
        # v2's pre-read gap: the new tests against the pre-fix retry (5bef042).
        old = subprocess.run(["git", "show", "5bef042:students/item_retry.py"], cwd=W,
                             capture_output=True, text=True, check=True).stdout
        (W / "students/item_retry.py").write_text(old)
        test(["students.tests_credits_mid_batch.AnyRetryInAStoppedBatchResumesIt"], "repro_fix_item_retry_5bef042")
    elif mode == "modules":
        test(MODULES, "changed_modules_and_guards")
    elif mode == "mutants":
        for name, (f, a, b) in MUTANTS.items():
            text = (W / f).read_text()
            assert text.count(a) == 1, name
            (W / f).write_text(text.replace(a, b))
            try:
                rc = test(NEW + ["students.tests_item_retry"], "mutant_" + name.split()[0], "s7c_settings_mut")
                print(name, "KILLED" if rc else "SURVIVED", flush=True)
            finally:
                (W / f).write_text(text)
    elif mode == "regression":
        # Rule 15 addendum 1: a model change regresses every app that reads it.
        # 0b's format: a pg_stat_activity snapshot first (no query text), then
        # -v 2 output with a wall-clock stamp on every line.
        snap = subprocess.run(
            [PY, "-c",
             "import django,os;os.environ.setdefault('DJANGO_SETTINGS_MODULE','settings_worktree');"
             "django.setup();from django.db import connection as c;cur=c.cursor();"
             "cur.execute(\"SELECT datname, state, wait_event_type, query_start FROM pg_stat_activity "
             "WHERE datname IS NOT NULL ORDER BY datname\");"
             "[print(*r, sep=' | ') for r in cur.fetchall()]"],
            cwd=W, capture_output=True, text=True)
        (LOGS / ("regression_pg_stat_activity_before_" + TAG + ".txt")).write_text(snap.stdout + snap.stderr)
        # 0b's command: flock on the machine lock, timeout 1800.
        cmd = ["nice", "-n", "10", "flock", os.path.expanduser("~/.machine-fullsuite.lock"),
               "timeout", "-k", "60", "1800", PY, "manage.py", "test",
               "--settings=settings_worktree", "--noinput", "-v", "2", *APPS]
        log = LOGS / ("regression_" + "_".join(APPS) + "_" + TAG + ".log")
        with log.open("w") as out:
            env = {k: v for k, v in os.environ.items()
                   if not k.startswith("RACE_COST_") and k != "AUDIT_BENCH"}
            proc = subprocess.Popen(cmd, cwd=W, stdout=subprocess.PIPE, stderr=subprocess.STDOUT,
                                    text=True, env=env)
            for line in proc.stdout:
                out.write(time.strftime("%H:%M:%S ") + line)
                out.flush()
            rc = proc.wait()
        keep = [l for l in log.read_text().splitlines() if l[9:].startswith(("Ran ", "OK", "FAILED", "FAIL:", "ERROR:"))]
        print(f"=== regression: exit {rc}\n" + "\n".join(keep), flush=True)
finally:
    if mode in ("repro", "repro_fix"):
        subprocess.run(["git", "checkout", "--", *PROD], cwd=W)
    assert sha() == before, "restore failed"
    print("restored ok", flush=True)
