def run(name, path, mutate, tests, frag):
    src = open(path).read(); h = hashlib.sha256(src.encode()).hexdigest()
    try:
        new = mutate(src)
        assert new != src, "mutation did not apply"
        open(path,"w").write(new)
        r = subprocess.run([sys.executable,"manage.py","test",*tests,"--settings=settings_worktree","--noinput"],capture_output=True,text=True)
        txt = r.stdout + r.stderr
        killed = r.returncode != 0
        fragok = frag in txt
        status = "KILLED" if killed else "SURVIVED"
        line = f"{name}: {status}; fragment {'MATCH' if fragok else 'MISMATCH'} ({frag!r})"
        if not fragok:
            errs = [l for l in txt.splitlines() if l.startswith(("AssertionError","RuntimeError","ValueError","AttributeError"))][:2]
            line += " | got: " + " / ".join(errs)
    finally:
        open(path,"w").write(src)
    ok = hashlib.sha256(open(path).read().encode()).hexdigest() == h
    line += f"; restored sha256 {'match' if ok else 'MISMATCH'}"
    print(line); out.write(line+"\n"); out.flush()
