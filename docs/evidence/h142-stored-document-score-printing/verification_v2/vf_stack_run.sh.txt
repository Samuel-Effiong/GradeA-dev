#!/bin/bash
# v2's one slot for the score-printing stack (tip 67390ac8, code c8ec622d). Only after the Release Engineer's GRANT.
# Usage: vf_stack_run.sh <tip-sha8>   (start by full path)
set -uo pipefail
if [ -z "${GAP_INHIBITED:-}" ]; then GAP_INHIBITED=1 exec systemd-inhibit --what=idle:sleep:handle-lid-switch --who=GAP --why=GAP-test-run --mode=block "$0" "$@"; fi
TIP=$1
P=/home/bond-servant-in-training/Documents/Projects
WT=$P/Grade-Automator-Plus-vf2-s1
H=$P/GAP-v2-handover
PY=/home/bond-servant-in-training/Documents/Virtualenvs/AutoGrader_env/bin/python
FILES="students/tests_vf2_score_probe.py students/tests_vf2_h144_probe.py students/tests_vf2_h145_probe.py students/tests_vf2_h146_probe.py students/tests_vf2_g_probe.py"
SRC_score=tests_vf2_score_probe.py; SRC_h144=tests_vf2_h144_probe.py; SRC_h145=tests_vf2_h145_probe.py; SRC_h146=tests_vf2_h146_probe.py; SRC_g=tests_vf2_g_probe.py
export COMMIT=$TIP MUTANT_LOGS=$H/runs/stack_${TIP}_mutant_logs EXEMPT_EMAIL_DOMAINS= PYTHONDONTWRITEBYTECODE=1
ST=$H/runs/stack_${TIP}.status
WRAP="systemd-run --user --scope -q -p MemoryMax=6G -p MemorySwapMax=0 nice -n 10 timeout -k 60 1800"
say() { echo "$(date +%H:%M:%S) $*" >> $ST; }
cd $WT || exit 9
[ "$(git rev-parse --short=8 HEAD)" = "$TIP" ] && [ -z "$(git status --porcelain)" ] || { echo "NOT FROZEN at $TIP"; exit 9; }
[ ! -e $MUTANT_LOGS ] || { echo "logs of $TIP exist already"; exit 9; }
for pair in "$SRC_score:students/tests_vf2_score_probe.py" "$SRC_h144:students/tests_vf2_h144_probe.py" "$SRC_h145:students/tests_vf2_h145_probe.py" "$SRC_h146:students/tests_vf2_h146_probe.py" "$SRC_g:students/tests_vf2_g_probe.py"; do
  cp $H/${pair%%:*} $WT/${pair##*:}
  cmp -s $H/${pair%%:*} $WT/${pair##*:} || { echo "probe copy differs: ${pair##*:}"; for f in $FILES; do rm -f $WT/$f; done; exit 9; }
done
EXPECT=$(for f in $FILES; do echo "?? $f"; done | sort)
[ "$(git status --porcelain | sort)" = "$EXPECT" ] || { echo "unexpected files"; for f in $FILES; do rm -f $WT/$f; done; exit 9; }
finish() {
  for f in $FILES; do rm -f $WT/$f; done
  rm -rf $WT/students/__pycache__ $WT/assignments/__pycache__
  if [ -z "$(git -C $WT status --porcelain)" ] && [ "$(git -C $WT rev-parse --short=8 HEAD)" = "$TIP" ]; then say "probes removed; tree is the frozen tip again"; else say "WARNING: tree not the frozen tip at exit"; fi
}
trap finish EXIT
for step in check baseline mutants; do
  say "$step start: loadavg $(cat /proc/loadavg)"
  rc=0; $WRAP $PY -B $H/vf_stack_mutants.py $step > $H/runs/stack_${TIP}_$step.log 2>&1 < /dev/null || rc=$?
  say "$step end exit=$rc: loadavg $(cat /proc/loadavg)"
  [ $step = check ] && [ $rc -ne 0 ] && { echo "check failed"; exit 1; }
  [ $step = baseline ] && [ $rc -ne 0 ] && { echo "baseline red: no mutant is run"; exit 1; }
done
echo "done: see $ST"
