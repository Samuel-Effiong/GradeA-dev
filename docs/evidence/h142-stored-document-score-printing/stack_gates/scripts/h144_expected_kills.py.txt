#: the red commit (the tests, no fix): everything but the three controls.
REPRO = {O_CLEARS, O_ANSWER, O_STUDENT, U_CLEARS, W_FAILED, Q_BROKER, Q_OTHER, Q_LOG, Q_STUDENT}
LINE = re.compile(r"^(?:FAIL|ERROR): (test_\w+) \(([\w.]+)\)", re.M)


def failing(text):
    return {(path.split(".")[-2], name) for name, path in LINE.findall(text)}


def compare(label, got, expected):
    extra, missing = got - expected, expected - got
    print(
        f"{label}: {len(got)} failing, {'as expected' if got == expected else 'DIFFERS'}"
        + (f"; unexpected: {sorted(extra)}" if extra else "")
        + (f"; expected but passed: {sorted(missing)}" if missing else "")
    )
    return got != expected


if sys.argv[1] == "--repro":
    wanted = REPRO
    bad = compare("repro", failing(Path(sys.argv[2]).read_text(errors="replace")), wanted)
    print("the red set is the expected one" if not bad else "the red set differs")
else:
    logs = Path(sys.argv[1]) / "logs"
    bad = sum(
        compare(mutant, failing((logs / f"{mutant}.log").read_text(errors="replace")), expected)
        for mutant, expected in EXPECTED.items()
    )
    print("every mutant's failing set is the expected one" if not bad else f"{bad} mutant(s) differ")
sys.exit(1 if bad else 0)
