#: the second red commit, 4e22fda5: its rule is c9ae46b3's, so the four new
#: "follows" tests fail, with both samples each, and nothing else
REPRO = WRAPPED | MADE | DECORATORS | {LAMBDA, LAMBDA_HELPER}
LINE = re.compile(r"^(?:FAIL|ERROR): (test_\w+) \([\w.]+\)(?: \(shape=(['\"])(.*)\2\))?$", re.M)


def failing(text):
    return {(name, sample or None) for name, _quote, sample in LINE.findall(text)}


def compare(label, got, expected):
    extra, missing = got - expected, expected - got
    print(
        f"{label}: {len(got)} failing, {'as expected' if got == expected else 'DIFFERS'}"
        + (f"; unexpected: {sorted(extra, key=str)}" if extra else "")
        + (f"; expected but passed: {sorted(missing, key=str)}" if missing else "")
    )
    return got != expected


if sys.argv[1] == "--repro":
    bad = compare("repro", failing(Path(sys.argv[2]).read_text(errors="replace")), REPRO)
    print("the red set is the expected one" if not bad else "the red set differs")
else:
    logs = Path(sys.argv[1]) / "logs"
    bad = sum(
        compare(mutant, failing((logs / f"{mutant}.log").read_text(errors="replace")), expected)
        for mutant, expected in EXPECTED.items()
    )
    print("every mutant's failing set is the expected one" if not bad else f"{bad} mutant(s) differ")
sys.exit(1 if bad else 0)
