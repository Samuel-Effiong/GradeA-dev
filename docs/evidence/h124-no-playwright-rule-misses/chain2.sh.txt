#!/bin/bash
# H-124 gate chain, SECOND RUN (after v2's pre-read), on the frozen tip: (r) the red commit's module in a
# disposable worktree, expected red; (a) the changed guard module with the
# repo-wide guard list, serial; (b) the battery. (c) is c_h124.sh, on its
# own grant. Rule 18: every test command writes straight to a file, stdin
# from /dev/null. No wall-clock assertion is in these runs (the renderer's
# real-browser modules are not in them); load is recorded all the same.
# Usage: chain.sh <tip-sha8> <red-sha8>
set -euo pipefail
TIP=$1; RED=$2
P=/home/bond-servant-in-training/Documents/Projects
WT=$P/Grade-Automator-Plus-h124-no-playwright-rule-misses
RW=$P/Grade-Automator-Plus-h124-repro
OUT=$P/GAP-d5-runs/h124
E=docs/evidence/h124-no-playwright-rule-misses
export EXEMPT_EMAIL_DOMAINS=
WRAP="systemd-inhibit --what=idle:sleep:handle-lid-switch --who=GAP --why=GAP-test-run --mode=block systemd-run --user --scope -q -p MemoryMax=6G -p MemorySwapMax=0 nice -n 10 timeout -k 60 1800"
say() { echo "$(date +%H:%M:%S) $*" | tee -a $OUT/chain.status; }
load() { say "$1: loadavg $(cat /proc/loadavg)"; }
cd $WT
[ "$(git rev-parse --short=8 HEAD)" = "$TIP" ] && [ -z "$(git status --porcelain)" ] || { say "ABORT: tree not frozen at $TIP"; exit 9; }
[ ! -e $OUT/r_repro_$RED.log ] && [ ! -e $OUT/a_modules_$TIP.log ] && [ ! -e $OUT/battery_$TIP ] || { say "ABORT: logs of $TIP or $RED exist already"; exit 9; }
[ ! -e $RW ] || { say "ABORT: $RW exists already"; exit 9; }
BATTERY_RAN=0
tidy() {
  set +e
  cd $WT
  if [ -e $RW ]; then git worktree remove --force $RW; git worktree prune; fi
  if [ $BATTERY_RAN = 1 ] && { [ -e $E/logs ] || [ -e $E/results.tsv ]; }; then
    mkdir -p $OUT/battery_$TIP && mv $E/logs $E/results.tsv $OUT/battery_$TIP/ || say "WARNING: could not move the battery's output"
  fi
  [ -z "$(git status --porcelain)" ] && [ "$(git rev-parse --short=8 HEAD)" = "$TIP" ] || say "WARNING: tree not the frozen tip $TIP at exit"
}
trap tidy EXIT

# (r) the red commit: the guard module alone, expected exit 1 with the red set
git worktree add --detach $RW $RED > /dev/null 2>&1
ln -s $P/Grade-Automator-Plus/.env $RW/.env
printf 'from AutoGrader.settings import *  # noqa: F401,F403\nfrom AutoGrader.settings import DATABASES\n\nDATABASES["default"].setdefault("TEST", {})\nDATABASES["default"]["TEST"]["NAME"] = "test_h124_repro"\n' > $RW/settings_worktree.py
load "(r) start"; s=$(date +%s); rc=0
( cd $RW && PYTHONDONTWRITEBYTECODE=1 $WRAP python manage.py test AutoGrader.tests_no_playwright_at_import --settings=settings_worktree --noinput > $OUT/r_repro_$RED.log 2>&1 < /dev/null ) || rc=$?
echo "exit=$rc wall_s=$(( $(date +%s)-s ))" >> $OUT/r_repro_$RED.log
load "(r) end"
say "(r) exit=$rc (1 is the expected one)"
git worktree remove --force $RW; git worktree prune
[ $rc -eq 1 ] || { say "STOP: (r) did not end red in the ordinary way"; exit 1; }
rc2=0; python3 $OUT/expected_kills_2.py --repro $OUT/r_repro_$RED.log > $OUT/expected_repro_$RED.txt 2>&1 || rc2=$?
say "(r) $(tail -n 1 $OUT/expected_repro_$RED.txt)"
[ $rc2 -eq 0 ] || { say "STOP: the red set differs from the expected one"; exit 1; }

# (a) the changed guard module and the repo-wide guard list
load "(a) start"; s=$(date +%s); rc=0
$WRAP python manage.py test AutoGrader.tests_no_playwright_at_import AutoGrader.tests_no_wildcard_invalidation AutoGrader.tests_cache_invalidation_coverage AutoGrader.tests_migration_rollback_defaults AutoGrader.tests_redis_test_isolation AutoGrader.tests_beat_health AutoGrader.tests_beat_locks AutoGrader.tests_management_commands_are_commands AutoGrader.tests_error_messages AutoGrader.tests_no_pii_in_logs AutoGrader.tests_log_scrubbing AutoGrader.tests_sentry_scrubbing AutoGrader.tests_redis_hygiene_databases AutoGrader.tests_redis_hygiene AutoGrader.tests_patient_test_stream AutoGrader.tests_pool_worker_sigterm billing.tests.test_logs_carry_no_email billing.tests.test_log_scrubbing_end_to_end classrooms.tests_teacher_access_sweep classrooms.tests_course_roster_scope_sweep assignments.tests_schema_extension users.tests_schema_extension --settings=settings_worktree --noinput --verbosity 2 > $OUT/a_modules_$TIP.log 2>&1 < /dev/null || rc=$?
echo "exit=$rc wall_s=$(( $(date +%s)-s ))" >> $OUT/a_modules_$TIP.log
load "(a) end"
say "(a) exit=$rc"; [ $rc -eq 0 ] || { say "STOP: (a) red"; exit 1; }

# (b) the battery
load "(b) start"; s=$(date +%s); rc=0; BATTERY_RAN=1
PYTHONDONTWRITEBYTECODE=1 $WRAP python $E/run_mutants.py $TIP > $OUT/b_mutation_battery_$TIP.log 2>&1 < /dev/null || rc=$?
echo "exit=$rc wall_s=$(( $(date +%s)-s ))" >> $OUT/b_mutation_battery_$TIP.log
load "(b) end"
say "(b) exit=$rc"
tidy; BATTERY_RAN=0
rc2=0; python3 $OUT/expected_kills_2.py $OUT/battery_$TIP > $OUT/expected_kills_$TIP.txt 2>&1 || rc2=$?
say "expected failing sets: $(tail -n 1 $OUT/expected_kills_$TIP.txt)"
[ $rc -eq 0 ] || { say "STOP: (b) not all killed (exit 124 or 137 is a hang or a cap, reported as such)"; exit 1; }
[ $rc2 -eq 0 ] || { say "STOP: a mutant's failing set differs from the expected one"; exit 1; }
say "H-124 CHAIN END: (c) needs its own GRANT"
