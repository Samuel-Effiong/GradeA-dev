"""Item 6 in-repo mutants (Builder 2 · 4d): deliberate breaks, and the tests that must catch each.

Run by gate.py inside the throwaway worktree item6-mut (cwd), copied there as
item6_mutate.py; `python -I mutate.py --check` in any tree only counts the
anchors (no run). Each mutant: the file is edited (anchor found exactly once),
the mutated module's __pycache__ removed (rule 17), its labels run with
--settings=settings_worktree --keepdb (after one baseline that creates the
database), output straight to its own log file with stdin from /dev/null
(rule 18), the file restored from the saved source and sha256-checked.

Judged (rules 18, 19, 22; H-229): KILLED AS WRITTEN = non-zero exit, the run's
own "Ran" line, no load failure, the failing set EXACTLY the written one, and
each written fragment inside that test's own failure block, all of a test's
fragments in ONE block. Anything else is reported as what it is (SURVIVED,
DIFFERS with the extra or missing tests named, WRONG REASON, BROKEN) and the
battery ends non-zero. EXPECTED was written before any run.
"""

#: Written before any run: {mutant: {test: [fragments all in ONE of its blocks]}}.
EXPECTED = {

    # which first fails at its error-text assertion with both sentences in
    # the block (H-225's chain config, P1). C1/C2 and the other 19 unit tests

    "M228-1": {

    },

LINE = re.compile(r"^(?:FAIL|ERROR): (test_\w+) \(([\w.]+)\)", re.M)
BLOCK = re.compile(
    r"^(?:FAIL|ERROR): (test_\w+) \(([\w.]+)\)([^\n]*\n.*?)(?=^(?:FAIL|ERROR): |^Ran \d+ tests)",
    re.M | re.S,
)
RAN = re.compile(r"^Ran (\d+) tests? in ", re.M)
LOAD_FAILURE = ("unittest.loader._FailedTest", "ModuleNotFoundError", "ImportError:")


def sha(path):
    return hashlib.sha256(pathlib.Path(path).read_bytes()).hexdigest()


def clear_pycache(rel):
    shutil.rmtree(pathlib.Path(rel).parent / "__pycache__", ignore_errors=True)


def run_tests(labels, log):
    with open(log, "w") as fh, open(os.devnull) as nothing:
        rc = subprocess.run(
            [sys.executable, "manage.py", "test", "--noinput", "--verbosity", "2",
             f"--settings={SETTINGS}", "--keepdb", *labels],
            stdin=nothing, stdout=fh, stderr=subprocess.STDOUT,
            env={**os.environ, "PYTHONDONTWRITEBYTECODE": "1", "EXEMPT_EMAIL_DOMAINS": ""},
        ).returncode
    return rc, open(log, errors="replace").read()


def judge(mid, rc, text):
    expected = EXPECTED[mid]
    problems = []
    if rc == 0:
        problems.append("SURVIVED (exit 0)")
    if not RAN.search(text):
        problems.append("no Ran line")
    if any(s in text for s in LOAD_FAILURE):
        problems.append("load failure")
    failed = {name for name, _ in LINE.findall(text)}
    extra, missing = sorted(failed - set(expected)), sorted(set(expected) - failed)
    if extra or missing:
        problems.append(f"DIFFERS: also failed {extra}; did not fail {missing}")
    blocks = {}
    for name, _, body in BLOCK.findall(text):
        blocks.setdefault(name, []).append(body)
    for test, fragments in expected.items():
        bodies = blocks.get(test, [])
        if test in failed and not any(all(f in b for f in fragments) for b in bodies):
            problems.append(f"WRONG REASON {test}: {fragments} not together in one of its blocks")
    return problems, sorted(failed)


def main():
    if "--check" in sys.argv:
        bad = 0
        for mid, (_, rel, old, _new, _labels) in MUTANTS.items():
            n = pathlib.Path(rel).read_text().count(old)
            print(f"{mid}: anchor in {rel} found {n} time(s)")
            bad += n != 1
        print("anchors ok" if not bad else f"{bad} anchor(s) wrong")
        return 1 if bad else 0

    LOGS.mkdir(parents=True, exist_ok=True)
    labels = sorted({l for m in MUTANTS.values() for l in m[4]})
    rc, text = run_tests(labels, LOGS / "baseline.log")
    m = RAN.search(text)
    print(f"baseline: exit={rc} {m.group(0).strip() if m else 'NO Ran LINE'}", flush=True)
    if rc != 0 or not m:
        print("BATTERY: the baseline is not green; no mutant run")
        return 2
    rows = {}
    bad = 0
    for mid, (what, rel, old, new, mlabels) in MUTANTS.items():
        path = pathlib.Path(rel)
        source = path.read_text()
        before = sha(path)
        if source.count(old) != 1:
            print(f"{mid}: BROKEN, anchor found {source.count(old)} times")
            bad += 1
            continue
        path.write_text(source.replace(old, new))
        clear_pycache(rel)
        try:
            rc, text = run_tests(mlabels, LOGS / f"{mid}.log")
        finally:
            path.write_text(source)
            clear_pycache(rel)
            if sha(path) != before:
                print(f"{mid}: RESTORE FAILED its sha check; stopping")
                return 3
        problems, failed = judge(mid, rc, text)
        m = RAN.search(text)
        verdict = "KILLED as written" if not problems else "; ".join(problems)
        print(f"{mid} ({what}): exit={rc} {m.group(0).strip() if m else 'no Ran'}; failing {failed}; {verdict}", flush=True)
        rows[mid] = {"what": what, "file": rel, "exit": rc, "failed": failed,
                     "expected": EXPECTED[mid], "verdict": verdict}
        bad += bool(problems)
    RESULTS.write_text(json.dumps(rows, indent=2))
    if bad:
        print(f"BATTERY: {bad} mutant(s) not killed as written")
        return 1
    print("BATTERY: every mutant killed as written")
    return 0


if __name__ == "__main__":
    sys.exit(main())
