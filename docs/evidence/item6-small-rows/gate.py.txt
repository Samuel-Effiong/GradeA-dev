Every test command: its own systemd scope (MemoryMax = available - 2G at start,
min 6G, else STOP exit 5; MemorySwapMax=0), nice 10, timeout 1800 (-k 60),

unit name) when available memory falls under 1.5 GB; a stall watchdog stops

Judging (rules 19, 22, H-229): a red step needs a non-zero exit, the run's
own "Ran N tests" line with the written N, no OK line, no load failure, the

class Stop(Exception):
    pass


def now():
    return time.strftime("%H:%M:%S")


def avail_kb():
    for line in open("/proc/meminfo"):
        if line.startswith("MemAvailable:"):
            return int(line.split()[1])
    raise Stop("MemAvailable not found")


class Gate:
    def __init__(self, tip, h120):
        self.tip = subprocess.run(
            ["git", "rev-parse", "--short=8", tip], cwd=WT, check=True, capture_output=True, text=True
        ).stdout.strip()
        self.h120 = h120
        self.out = HERE / f"gate_{self.tip}"
        self.out.mkdir(parents=True, exist_ok=True)
        self.report = open(self.out / "gate.out", "a")
        self.unit_prefix = f"gap-item6-{os.getpid()}"
        self.part = "none"
        self.stop_watchdogs = False
        avail_g = avail_kb() // 1048576
        self.cap = max(avail_g - 2, 0)
        self.say(f"item 6 gate on {self.tip}, base {BASE}: {time.strftime('%F %T')}; loadavg {open('/proc/loadavg').read().strip()}")
        self.say(f"available at start: {avail_g}G; cap {self.cap}G")
        if self.cap < 6:
            raise Stop("cap under 6G (exit 5): wait and tell the Senior Manager")

    def say(self, text):
        line = f"[{now()}] {text}"
        print(line, flush=True)
        self.report.write(line + "\n")
        self.report.flush()

    # ---- running things -------------------------------------------------
    def unit(self):
        return f"{self.unit_prefix}-{self.part}"

    def wrap(self, timeout=TIMEOUT):
        return [
            "systemd-run", "--user", "--scope", "-q", f"--unit={self.unit()}",
            "-p", f"MemoryMax={self.cap}G", "-p", "MemorySwapMax=0",
            "nice", "-n", "10", "timeout", "-k", "60", str(timeout),
        ]

    def run(self, cmd, log, part, cwd, env=None, timeout=TIMEOUT):
        self.part = part
        full_env = {**os.environ, "PYTHONFAULTHANDLER": "1", "EXEMPT_EMAIL_DOMAINS": "", **(env or {})}
        self.say(f"{part}: {' '.join(str(c) for c in cmd)} -> {log.name}")
        with open(log, "w") as fh, open(os.devnull) as nothing:
            rc = subprocess.run(
                self.wrap(timeout) + [str(c) for c in cmd], cwd=cwd, stdin=nothing,
                stdout=fh, stderr=subprocess.STDOUT, env=full_env,
            ).returncode
        text = open(log, errors="replace").read()
        ran = RAN.search(text)
        self.say(f"{part}: exit={rc} {ran.group(0).strip() if ran else 'NO Ran LINE'}")
        self.part = "none"
        return rc, text

    def django_test(self, tree, labels, log, part, settings="settings_worktree", keepdb=False):
        cmd = [PY, "manage.py", "test", "--noinput", "--verbosity", "2", f"--settings={settings}", *labels]
        if keepdb:
            cmd.append("--keepdb")
        return self.run(cmd, log, part, cwd=tree, env={"PYTHONDONTWRITEBYTECODE": "1"})

    # ---- watchdogs -------------------------------------------------------
    def memory_watchdog(self):
        while not self.stop_watchdogs:
            time.sleep(10)
            if avail_kb() < 1572864 and self.part != "none":
                unit = self.unit() + ".scope"
                with open(self.out / "memory_watchdog.txt", "a") as fh:
                    fh.write(f"MEMORY STOP {now()}: available {avail_kb()} kB; stopping {unit}\n")
                subprocess.run(["systemctl", "--user", "stop", unit])

    def stall_watchdog(self, logs_dir):
        time.sleep(60)
        while self.part == "mutants":
            newest = max(logs_dir.glob("*.log"), key=lambda p: p.stat().st_mtime, default=None)
            if newest and time.time() - newest.stat().st_mtime > 300:
                unit = self.unit() + ".scope"
                with open(self.out / "stall_watchdog.txt", "a") as fh:
                    fh.write(f"STALL STOP {now()}: {newest.name} silent over 300 s; stopping {unit}\n")
                subprocess.run(["systemctl", "--user", "stop", unit])
                break
            time.sleep(15)

    # ---- worktrees -------------------------------------------------------
    def add_tree(self, name, commit, db):
        path = P / f"Grade-Automator-Plus-{name}"
        if path.exists():
            raise Stop(f"{path} exists already: not mine to remove")
        subprocess.run(["git", "worktree", "add", "--detach", str(path), commit], cwd=WT, check=True, capture_output=True)
        if not (path / ".env").exists():
            os.symlink(P / "Grade-Automator-Plus" / ".env", path / ".env")
        self.settings(path, db)
        return path

    def settings(self, tree, db):
        (tree / "settings_worktree.py").write_text(
            '"""Item 6 gate: a unique test database for this throwaway tree."""\n'
            "from AutoGrader.settings import *  # noqa: F401,F403\n"
            "from AutoGrader.settings import DATABASES\n\n"
            'DATABASES["default"].setdefault("TEST", {})\n'
            f'DATABASES["default"]["TEST"]["NAME"] = "{db}"\n'
        )

    def remove_tree(self, path, written=()):
        for rel in ("settings_worktree.py", ".env", *written):
            f = path / rel
            if f.is_symlink() or f.exists():
                f.unlink()
        for cache in path.rglob("__pycache__"):
            shutil.rmtree(cache, ignore_errors=True)
        r = subprocess.run(["git", "worktree", "remove", str(path)], cwd=WT, capture_output=True, text=True)
        if r.returncode:
            self.say(f"NOTE: git refused to remove {path} (no --force): {r.stderr.strip()[:200]}; left for the Senior Manager")
        else:
            self.say(f"removed {path.name}")

    def drop_db(self, tree, settings, db):
        code = (
            "from django.db import connection\n"
            "with connection._nodb_cursor() as c:\n"
            f"    c.execute('DROP DATABASE IF EXISTS \"{db}\"')\n"
            f"print('dropped {db}')\n"
        )
        with open(self.out / f"drop_{db}.txt", "w") as fh, open(os.devnull) as nothing:
            subprocess.run([PY, "manage.py", "shell", f"--settings={settings}", "-c", code], cwd=tree, stdin=nothing, stdout=fh, stderr=subprocess.STDOUT)

    # ---- judging ---------------------------------------------------------
    @staticmethod
    def failing(text):
        return {name for name, _ in LINE.findall(text)}

    @staticmethod
    def blocks(text):
        out = {}
        for name, _, body in BLOCK.findall(text):
            out.setdefault(name, []).append(body)
        return out

    def judge_red(self, label, rc, text, expected, ran):
        bad = []
        if rc == 0:
            bad.append("exit 0")
        m = RAN.search(text)
        if not m:
            bad.append("no Ran line")
        elif int(m.group(1)) != ran:
            bad.append(f"Ran {m.group(1)}, written {ran}")
        if re.search(r"^OK", text, re.M):
            bad.append("an OK line")
        if any(s in text for s in LOAD_FAILURE):
            bad.append("a load failure")
        got = self.failing(text)
        if got != set(expected):
            bad.append(f"failing set {sorted(got)} != written {sorted(expected)}")
        blocks = self.blocks(text)
        for test, fragments in expected.items():
            bodies = blocks.get(test, [])
            if not bodies:
                bad.append(f"{test}: no failure block parsed")
            elif not any(all(f in body for f in fragments) for body in bodies):
                bad.append(f"{test}: WRONG REASON, written {fragments} not together in one of its blocks")
        if bad:
            raise Stop(f"{label}: not red as written: {'; '.join(bad)}")
        self.say(f"{label}: red as written ({len(expected)} tests, each for its written reason)")

    def judge_green(self, label, rc, text, ran=None):
        m = RAN.search(text)
        ok = rc == 0 and m and re.search(r"^OK", text, re.M) and not any(s in text for s in LOAD_FAILURE)
        if ok and ran is not None and int(m.group(1)) != ran:
            ok = False
        if not ok:
            raise Stop(f"{label}: not green (exit {rc}, {m.group(0).strip() if m else 'no Ran line'})")
        self.say(f"{label}: green ({m.group(0).strip()})")

    # ---- the steps -------------------------------------------------------
    def preflight(self):
        head = subprocess.run(["git", "rev-parse", "--short=8", "HEAD"], cwd=WT, capture_output=True, text=True).stdout.strip()
        if head != self.tip:
            raise Stop(f"the worktree is at {head}, not {self.tip}")
        dirty = [l for l in subprocess.run(["git", "status", "--porcelain"], cwd=WT, capture_output=True, text=True).stdout.splitlines() if "settings_worktree" not in l]
        if dirty:
            raise Stop(f"the worktree is not clean: {dirty[:5]}")
        for name in ("item6-redt", "item6-gap", "item6-redc", "item6-mut"):
            if (P / f"Grade-Automator-Plus-{name}").exists():
                raise Stop(f"Grade-Automator-Plus-{name} exists already")
        self.settings(WT, "test_item6_small_rows")
        for f, sha in (("gate.py", None), ("mutate.py", None)):
            digest = hashlib.sha256((HERE / f).read_bytes()).hexdigest()
            self.say(f"{f} sha256 {digest}")
        self.say(f"row_chain.py sha256 {hashlib.sha256((P / 'GAP-58-runs' / 'row_chain.py').read_bytes()).hexdigest()}")

    def step_0a(self):

        self.judge_red("0a H-226 at the tests commit", rc, text, RED_226, ran=12)

            self.remove_tree(tree)
        self.judge_green("0b H-228: P1 with the OLD test (the gap, seen)", rc, text, ran=20)

        self.say(f"2b: H-120 harness exit={rc}; tail:\n{tail}")
        if rc != 0:
            raise Stop("2b: H-120's harness did not end green")


def main():
    args = sys.argv[1:]
    h120 = "--h120" in args
    tip = [a for a in args if not a.startswith("--")][0]
    gate = Gate(tip, h120)
    threading.Thread(target=gate.memory_watchdog, daemon=True).start()
    try:
        gate.preflight()
        gate.step_0a()
        gate.step_0b()
        gate.step_0c()
        gate.step_0d()
        gate.step_1a()
        gate.step_1b()
        gate.step_1()
        gate.step_1c()
        gate.step_1d()
        gate.step_2()
        gate.step_2b()
    except Stop as stop:
        gate.say(f"STOP: {stop}")
        gate.say(f"GATE END (stopped) {time.strftime('%F %T')}; loadavg {open('/proc/loadavg').read().strip()}")
        return 4
    finally:
        gate.stop_watchdogs = True
    gate.say(f"GATE END: green as written {time.strftime('%F %T')}; loadavg {open('/proc/loadavg').read().strip()}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
