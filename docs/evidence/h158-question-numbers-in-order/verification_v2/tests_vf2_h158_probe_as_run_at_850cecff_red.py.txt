class VF2H158FieldProbe(TransactionTestCase):
    def test_z1_the_number_field_refuses_what_the_last_defence_never_sees(self):
        field = QuestionSerializer().fields["question_number"]

        self.assertEqual(field.run_validation("3"), 3)
        for bad in ([1], {"n": 1}, "2a", None, True, 2.5):
            with self.subTest(bad=repr(bad)):
                with self.assertRaises(drf.ValidationError):
                    field.run_validation(bad)


@override_settings(
    GRADING_SECOND_OPINION_ENABLED=False,
    GRADING_EVIDENCE_ENFORCEMENT=ai_services.MODE_LOG,
    GRADING_ANSWER_CACHE_ENABLED=True,
)
class VF2H158PipelineProbe(TransactionTestCase):
    def setUp(self):
        cache.clear()
        self.addCleanup(cache.clear)
        self.teacher = CustomUser.objects.create_user(
            email="vf2-h158-teacher@example.com", user_type=UserTypes.TEACHER
        )
        self.student = CustomUser.objects.create_user(
            email="vf2-h158-student@example.com", user_type=UserTypes.STUDENT
        )
        session = Session.objects.create(name="S", teacher=self.teacher)
        self.course = Course.objects.create(
            name="C", teacher=self.teacher, session=session
        )
        StudentCourse.objects.create(
            student=self.student,
            course=self.course,
            enrollment_status=EnrollmentStatusType.ENROLLED,
        )
        self.prompts = []

    def extracted_paper(self, model_numbers, answer_numbers=(1, 2, 3)):
        """The extraction runs for real (its AI call replaced); the saved
        assignment carries the questions it returned."""
        questions = []
        for position, number in enumerate(model_numbers, start=1):
            entry = d5t.essay(position)
            entry["question_number"] = number
            questions.append(entry)
        reply = extraction_payload(
            assignment_type="ESSAY",
            questions=questions,
            total_points=30,
            question_count=3,
        )
        with patch(EXTRACT, return_value=reply):
            data = AssignmentProcessingService.extract_assignment_data(
                self.teacher,
                [{"type": "text", "text": "extract this"}],
                course=self.course,
            )
        assignment = Assignment.objects.create(
            title="A paper",
            course=self.course,
            status=AssignmentStatus.PUBLISHED,
            total_points=30,
            questions=data["questions"],
        )
        submission = StudentSubmission.objects.create(
            assignment=assignment,
            student=self.student,
            answers=[d5t.answer(n) for n in answer_numbers],
        )
        return assignment, submission, data

    def grade(self, submission, evaluated=(1, 2, 3)):
        def provider(**kwargs):
            self.prompts.append(str(kwargs.get("user_prompt")))
            return d5t.reply(
                {
                    "question_evaluations": [
                        d5t.evaluation(number, score)
                        for number, score in zip(evaluated, (8, 0, 8), strict=True)
                    ]
                }
            )

        with (
            patch.object(AIProcessor, "execute_graded_task") as execute,
            patch("students.services._formatted_grade_task") as formatted,
            patch("students.services.student_summary_async"),
        ):
            execute.side_effect = provider
            formatted.return_value.delay.return_value = types.SimpleNamespace(
                id="vf2-fake-task-id"
            )
            grade_engine(self.teacher, StudentSubmission.objects.get(pk=submission.pk))
        return execute.call_count

    def assert_graded_in_full(self, model_numbers, expected=(1, 2, 3)):
        assignment, submission, data = self.extracted_paper(
            model_numbers, answer_numbers=expected
        )
        saved_assignment = Assignment.objects.get(pk=assignment.pk)
        self.assertEqual(
            [q["question_number"] for q in saved_assignment.questions],
            list(expected),
        )

        calls = self.grade(submission, evaluated=expected)

        self.assertGreaterEqual(calls, 1)
        saved = StudentSubmission.objects.get(pk=submission.pk)
        self.assertIsNotNone(saved.graded_at)
        self.assertEqual(saved.grading_state, GradingState.DONE)
        self.assertEqual(saved.max_points, 30)
        self.assertEqual(saved.score, Decimal("16"))
        self.assertNotIn(
            "answer_not_found", [reason["type"] for reason in saved.review_reasons]
        )
        self.assertTrue(self.prompts)
        self.assertIn(f"Essay {expected[2]} answer.", self.prompts[0])
        return data

    def test_z2_a_repeat_from_the_model_is_graded_in_full(self):
        self.assert_graded_in_full([1, 1, 2])

    def test_z3_control_a_paper_numbered_in_order(self):
        self.assert_graded_in_full([1, 2, 3])

    def test_z6_numbers_that_are_already_distinct_are_kept(self):
        data = self.assert_graded_in_full([5, 6, 7], expected=(5, 6, 7))

        self.assertEqual(
            data["ai_raw_payload"]["model_question_numbers"],
            {"5": "5", "6": "6", "7": "7"},
        )

    def test_z7_a_repeat_in_a_paper_that_does_not_start_at_one(self):
        self.assert_graded_in_full([5, 6, 6])
