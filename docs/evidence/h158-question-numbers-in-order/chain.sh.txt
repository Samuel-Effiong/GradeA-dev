#!/bin/bash
# H-158 gate chain on the frozen tip: (r) the row's test module at the
# tests-first commit, in a disposable worktree, expected red on a set
# written beforehand; (a) that module, the existing test modules that
# drive the AI assignment paths this row touches (extraction by text,
# edit and upload; generation and the draft save; AssignmentSerializer;
# the document builder; the grading pairing; from
# tree_search_af1c5b8c.txt) and the repo-wide guard list, serial. Left to
# the app regression: the cache-matrix and scale modules that name those
# routes. Not run anywhere: the tests_real_* modules, which make paid
# calls and are opt-in; (b) the battery of 18 mutants. (c) is
# c_h158.sh, on its own grant.
# Rule 18: every test command writes straight to a file, stdin from
# /dev/null. No wall-clock assertion is in these runs; load is recorded.
# Usage: chain.sh <tip-sha8>
set -euo pipefail
TIP=$1
P=/home/bond-servant-in-training/Documents/Projects
WT=$P/Grade-Automator-Plus-h158-question-numbers-in-order
RW=$P/Grade-Automator-Plus-h158-repro
OUT=$P/GAP-d5-runs/dupq
E=docs/evidence/h158-question-numbers-in-order
export EXEMPT_EMAIL_DOMAINS=
WRAP="systemd-inhibit --what=idle:sleep:handle-lid-switch --who=GAP --why=GAP-test-run --mode=block systemd-run --user --scope -q -p MemoryMax=6G -p MemorySwapMax=0 nice -n 10 timeout -k 60 1800"
say() { echo "$(date +%H:%M:%S) $*" | tee -a $OUT/chain.status; }
load() { say "$1: loadavg $(cat /proc/loadavg)"; }
cd $WT
[ "$(git rev-parse --short=8 HEAD)" = "$TIP" ] && [ -z "$(git status --porcelain)" ] || { say "ABORT: tree not frozen at $TIP"; exit 9; }
[ ! -e $OUT/r_repro_7be08a58.log ] && [ ! -e $OUT/a_modules_$TIP.log ] && [ ! -e $OUT/battery_$TIP ] || { say "ABORT: logs of this chain exist already"; exit 9; }
[ ! -e $RW ] || { say "ABORT: $RW exists already"; exit 9; }
BATTERY_RAN=0
tidy() {
  set +e
  cd $WT
  if [ -e $RW ]; then git worktree remove --force $RW; git worktree prune; fi
  if [ $BATTERY_RAN = 1 ] && { [ -e $E/logs ] || [ -e $E/results.tsv ]; }; then
    mkdir -p $OUT/battery_$TIP && mv $E/logs $E/results.tsv $OUT/battery_$TIP/ || say "WARNING: could not move the battery's output"
  fi
  [ -z "$(git status --porcelain)" ] && [ "$(git rev-parse --short=8 HEAD)" = "$TIP" ] || say "WARNING: tree not the frozen tip $TIP at exit"
}
trap tidy EXIT

repro() { # name commit label flag [file to take from the tip]
  local name=$1 commit=$2 label=$3 flag=$4 from_tip=${5:-} rc=0 rc2=0 s
  git worktree add --detach $RW $commit > /dev/null 2>&1
  if [ -n "$from_tip" ]; then git show $TIP:$from_tip > $RW/$from_tip; say "($name) $from_tip taken from $TIP, the rest is $commit"; fi
  ln -s $P/Grade-Automator-Plus/.env $RW/.env
  printf 'from AutoGrader.settings import *  # noqa: F401,F403\nfrom AutoGrader.settings import DATABASES\n\nDATABASES["default"].setdefault("TEST", {})\nDATABASES["default"]["TEST"]["NAME"] = "test_h158_repro"\n' > $RW/settings_worktree.py
  load "($name) start"; s=$(date +%s)
  ( cd $RW && PYTHONDONTWRITEBYTECODE=1 $WRAP python manage.py test $label --settings=settings_worktree --noinput > $OUT/${name}_repro_$commit.log 2>&1 < /dev/null ) || rc=$?
  echo "exit=$rc wall_s=$(( $(date +%s)-s ))" >> $OUT/${name}_repro_$commit.log
  load "($name) end"
  say "($name) exit=$rc (1 is the expected one)"
  git worktree remove --force $RW; git worktree prune
  [ $rc -eq 1 ] || { say "STOP: ($name) did not end red in the ordinary way"; exit 1; }
  python3 $OUT/expected_kills.py $flag $OUT/${name}_repro_$commit.log > $OUT/expected_${name}_$commit.txt 2>&1 || rc2=$?
  say "($name) $(tail -n 1 $OUT/expected_${name}_$commit.txt)"
  [ $rc2 -eq 0 ] || { say "STOP: ($name) the red set differs from the expected one"; exit 1; }
}
repro r 7be08a58 assignments.tests_question_numbers_in_order --repro assignments/tests_question_numbers_in_order.py

# (a)
load "(a) start"; s=$(date +%s); rc=0
$WRAP python manage.py test assignments.tests_question_numbers_in_order assignments.tests assignments.tests_extraction_service assignments.tests_ai_output_allowlist assignments.tests_course_ownership_idor assignments.tests_course_ownership_failure assignments.tests_partial_update_credit_gate assignments.tests_rigor assignments.tests_security assignments.tests_upload_batch_billing assignments.tests_upload_pipeline assignments.tests_upload_task_retry_policy assignments.tests_format_assignment_html billing.tests.test_execute_graded_task billing.tests.test_h38_part2_removed_teacher_routes billing.tests.test_refusal_handling students.tests_assignment_edit_notification ai_processor.tests_pdf_type_validation ai_processor.tests ai_processor.tests_grading_hardening AutoGrader.tests_cache_bespoke_1114 AutoGrader.tests_student_feedback_guard AutoGrader.tests_migration_safety_check AutoGrader.tests_no_playwright_at_import AutoGrader.tests_no_wildcard_invalidation AutoGrader.tests_cache_invalidation_coverage AutoGrader.tests_migration_rollback_defaults AutoGrader.tests_redis_test_isolation AutoGrader.tests_beat_health AutoGrader.tests_beat_locks AutoGrader.tests_management_commands_are_commands AutoGrader.tests_error_messages AutoGrader.tests_no_pii_in_logs AutoGrader.tests_log_scrubbing AutoGrader.tests_sentry_scrubbing AutoGrader.tests_redis_hygiene_databases AutoGrader.tests_redis_hygiene AutoGrader.tests_patient_test_stream AutoGrader.tests_pool_worker_sigterm billing.tests.test_logs_carry_no_email billing.tests.test_log_scrubbing_end_to_end classrooms.tests_teacher_access_sweep classrooms.tests_course_roster_scope_sweep assignments.tests_schema_extension users.tests_schema_extension --settings=settings_worktree --noinput --verbosity 2 > $OUT/a_modules_$TIP.log 2>&1 < /dev/null || rc=$?
echo "exit=$rc wall_s=$(( $(date +%s)-s ))" >> $OUT/a_modules_$TIP.log
load "(a) end"
say "(a) exit=$rc"; [ $rc -eq 0 ] || { say "STOP: (a) red"; exit 1; }

# (b) the battery
load "(b) start"; s=$(date +%s); rc=0; BATTERY_RAN=1
PYTHONDONTWRITEBYTECODE=1 $WRAP python $E/run_mutants.py $TIP > $OUT/b_mutation_battery_$TIP.log 2>&1 < /dev/null || rc=$?
echo "exit=$rc wall_s=$(( $(date +%s)-s ))" >> $OUT/b_mutation_battery_$TIP.log
load "(b) end"
say "(b) exit=$rc"
tidy; BATTERY_RAN=0
rc2=0; python3 $OUT/expected_kills.py $OUT/battery_$TIP > $OUT/expected_kills_$TIP.txt 2>&1 || rc2=$?
say "expected failing sets: $(tail -n 1 $OUT/expected_kills_$TIP.txt)"
[ $rc -eq 0 ] || { say "STOP: (b) not all killed (exit 124 or 137 is a hang or a cap, reported as such)"; exit 1; }
[ $rc2 -eq 0 ] || { say "STOP: a mutant's failing set differs from the expected one"; exit 1; }
say "H-158 CHAIN END: (c) needs its own GRANT"
