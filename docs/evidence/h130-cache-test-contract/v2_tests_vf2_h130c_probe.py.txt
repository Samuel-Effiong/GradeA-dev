import datetime

from django.core.cache import cache

import AutoGrader.tests_cache_bespoke_1114 as d5s
from AutoGrader.cache_generation import SCOPE_USER, get_generation
from classrooms.models import StudentCourse


class VF2SubmissionDetailProbe(d5s.SubmissionDetailFreshnessTests):
    def generation(self):
        return get_generation(SCOPE_USER, self.student.id)

    def read(self):
        data = self.get(self.student_client, self.url)
        self.assertIn("raw_input", data)
        self.assertIn("Student Information", data["raw_input"])
        return data

    def test_p1_the_released_students_equal_payload_is_a_fresh_one(self):
        self.get(self.teacher_client, self.url)  # materialises the snapshot
        self.submission.refresh_from_db()
        self.assertTrue(self.submission.raw_input)
        self.submission.is_published = True
        self.submission.save()
        before = self.read()
        self.assertIn("Original Title", before["raw_input"])
        generation = self.generation()

        self.retitle()

        self.assertNotEqual(self.generation(), generation)
        self.assertEqual(before, self.read())

    def test_p2_a_due_date_the_teacher_saves_reaches_the_unreleased_document(self):
        before = self.read()
        self.assertNotIn("2031-01-02", before["raw_input"])

        self.assignment.due_date = datetime.datetime(
            2031, 1, 2, 3, 4, 5, tzinfo=datetime.timezone.utc
        )
        self.assignment.save()

        after = self.read()
        self.assertIn("2031-01-02", after["raw_input"])
        before.pop("raw_input")
        after.pop("raw_input")
        self.assertEqual(before, after)

    def test_p3_the_known_limit_with_the_enrolment_row_deleted(self):
        deleted, _ = StudentCourse.objects.filter(
            student=self.student, course=self.course
        ).delete()
        self.assertEqual(deleted, 1)
        before = self.read()  # still 200: no enrolment check on this read
        self.assertIn("Original Title", before["raw_input"])
        generation = self.generation()

        self.retitle()

        self.assertEqual(self.generation(), generation)
        stale = self.read()
        self.assertIn("Original Title", stale["raw_input"])
        self.assertNotIn("Retitled By Teacher", stale["raw_input"])
        cache.clear()
        fresh = self.read()
        self.assertIn("Retitled By Teacher", fresh["raw_input"])

    def test_p4_three_declared_keys_are_not_in_the_students_answer(self):
        data = self.read()
        for key in ("id", "assignment", "score", "is_published"):
            self.assertIn(key, data)
        for key in ("assignment_title", "assignment_due_date", "course_title"):
            self.assertNotIn(key, data)
