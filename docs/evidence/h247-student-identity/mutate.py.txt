def show(path):
    return subprocess.run(
        ["git", "-C", ROOT, "show", f"{SHA}:{path}"], capture_output=True, check=True
    ).stdout


def sha(b):
    return hashlib.sha256(b).hexdigest()


def run_tests():
    p = subprocess.run(
        [
            "python3",
            "manage.py",
            "test",
            *MODULES,
            "--settings=settings_worktree",
            "--noinput",
            "--keepdb",
        ],
        cwd=ROOT,
        capture_output=True,
        text=True,
    )
    out = p.stdout + p.stderr
    failed = sorted(set(re.findall(r"^(?:FAIL|ERROR): (\w+)", out, re.M)))
    return p.returncode, failed, out


def main():
    only = sys.argv[1:]
    for _, path, _, old, _ in MUTANTS:
        assert show(path).decode().count(old) == 1, (path, old[:60])
    code, failed, out = run_tests()
    print("baseline", code, failed, flush=True)
    assert code == 0, out[-3000:]
    results = []
    for mid, path, what, old, new in MUTANTS:
        if only and mid not in only:
            continue
        original = show(path)
        full = f"{ROOT}/{path}"
        try:
            open(full, "w").write(original.decode().replace(old, new, 1))
            code, failed, out = run_tests()
            if "too many clients" in out:
                print(mid, "WARNING: too many clients in log", flush=True)
        finally:
            open(full, "wb").write(original)
        restored = open(full, "rb").read()
        verdict = "KILLED" if code != 0 else "SURVIVED"
        results.append(
            {
                "id": mid,
                "file": path,
                "what": what,
                "verdict": verdict,
                "failed": failed,
                "restored_sha256_match": sha(restored) == sha(original),
                "sha256": sha(restored),
            }
        )
        print(mid, verdict, len(failed), failed[:4], flush=True)
    json.dump(results, open(sys.argv[0].replace(".py", f"_{SHA}.json"), "w"), indent=1)
    status = subprocess.run(
        ["git", "-C", ROOT, "status", "--porcelain"], capture_output=True, text=True
    ).stdout
    print("porcelain after:", repr(status))


main()
