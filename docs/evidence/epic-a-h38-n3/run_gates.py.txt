"""H-38 N3 gates (author 1a): repro | modules | mutants | regression.

Usage: python -B run_gates.py <mode> <logs dir> [apps,comma,separated] [tag]
Every test run is wrapped per rules 16 (idle:sleep:handle-lid-switch), 13
(6G scope, no swap) and 12 (nice, timeout 1800), with --settings.
Rule 17: the mutants and their baseline run with PYTHONDONTWRITEBYTECODE=1,
and __pycache__ of the mutated modules' directories is deleted before the
baseline, before each mutant and after each restore.
"""

import hashlib
import os
import pathlib
import shutil
import subprocess
import sys
import time

W = pathlib.Path(
    "/home/bond-servant-in-training/Documents/Projects/Grade-Automator-Plus-epic-a-h38-n3"
)
HERE = pathlib.Path(__file__).resolve().parent
PY = "/home/bond-servant-in-training/Documents/Virtualenvs/AutoGrader_env/bin/python"
BASE = "3fff1382"  # phase2/epic-a (after the bundle 5 merge-down), the branch's base
PROD = ["assignments/tasks.py", "audit/enums.py", "AutoGrader/reason_codes.py"]
NEW = ["students.tests_h38_n3_refusal_audit"]
MUTANT_MODULES = NEW + ["AutoGrader.tests_reason_codes"]
MODULES = list(
    dict.fromkeys(
        NEW
        + [
            # the refusals' own tests, and every test that reads GRADING_FAILED
            "students.tests_h38_tasks_namespace",
            "students.tests_item_retry_h38",
            "assignments.tests_grading_audit_events",
            "students.tests_s6d_rubric_gate",
            "students.tests_s6d_provider_failure",
            # the catalogue and the emitter
            "AutoGrader.tests_reason_codes",
            "AutoGrader.tests_codederror_serialization",
            "AutoGrader.tests_error_messages",
            "audit.tests_enums",
            "audit.tests_emitter",
            "audit.tests_metadata",
            "audit.tests_metrics",
            "audit.tests_background_attribution",
            # ALL repo-wide guards (rule 15 addendum 2)
            "audit.tests_route_coverage",
            "audit.tests_history_guard",
            "audit.tests_state_change",
            "audit.tests_checks",
            "AutoGrader.tests_no_wildcard_invalidation",
            "AutoGrader.tests_cache_invalidation_coverage",
            "AutoGrader.tests_migration_rollback_defaults",
            "AutoGrader.tests_redis_test_isolation",
            # guards that arrived with the bundle 5 merge-down (3fff1382)
            "AutoGrader.tests_beat_locks",
            "AutoGrader.tests_beat_health",
            "AutoGrader.tests_management_commands_are_commands",
            "audit.tests_sweep_beat_lock",
            "billing.tests.test_logs_carry_no_email",
            "classrooms.tests_teacher_access_sweep",
            "classrooms.tests_course_roster_scope_sweep",
        ]
    )
)
WRAP = [
    "systemd-inhibit", "--what=idle:sleep:handle-lid-switch", "--who=GAP",
    "--why=GAP test run", "--mode=block",
    "systemd-run", "--user", "--scope", "-q", "-p", "MemoryMax=6G", "-p", "MemorySwapMax=0",
    "nice", "-n", "10",
]  # fmt: skip
ns = {}
exec((HERE / "h38n3_mutants.py").read_text(), ns)
MUTANTS = ns["MUTANTS"]


def clear_pycache():
    for d in sorted({str(pathlib.Path(f).parent) for f in PROD}):
        shutil.rmtree(W / d / "__pycache__", ignore_errors=True)


def test(labels, tag, settings="settings_worktree", no_bytecode=False):
    cmd = WRAP + [
        "timeout", "-k", "60", "1800", PY, "-B", "manage.py", "test",
        f"--settings={settings}", "--noinput", "-v", "2", *labels,
    ]  # fmt: skip
    env = {
        k: v for k, v in os.environ.items()
        if not k.startswith("RACE_COST_") and k != "AUDIT_BENCH"
    }  # fmt: skip
    env["PYTHONPATH"] = str(HERE)
    env["EXEMPT_EMAIL_DOMAINS"] = ""
    if no_bytecode:
        env["PYTHONDONTWRITEBYTECODE"] = "1"
    started = time.time()
    p = subprocess.run(cmd, cwd=W, capture_output=True, text=True, env=env)
    out = p.stdout + p.stderr + f"\nexit={p.returncode} wall_s={int(time.time() - started)}\n"
    (LOGS / f"{tag}.log").write_text(out)
    keep = [
        line for line in out.splitlines()
        if line.startswith(("Ran ", "OK", "FAILED", "FAIL:", "ERROR:"))
    ]  # fmt: skip
    print(f"=== {tag}: exit {p.returncode}\n" + "\n".join(keep), flush=True)
    return p.returncode, keep


def sha():
    return {f: hashlib.sha256((W / f).read_bytes()).hexdigest() for f in PROD}


def frozen():
    head = subprocess.run(
        ["git", "rev-parse", "--short=8", "HEAD"], cwd=W, capture_output=True, text=True
    ).stdout.strip()
    dirty = subprocess.run(
        ["git", "status", "--porcelain", "--untracked-files=no"],
        cwd=W, capture_output=True, text=True,
    ).stdout.strip()  # fmt: skip
    assert not dirty, f"tree not clean:\n{dirty}"
    return head


LOGS = pathlib.Path(sys.argv[2])
LOGS.mkdir(parents=True, exist_ok=True)
mode = sys.argv[1]
APPS = sys.argv[3].split(",") if len(sys.argv) > 3 else ["assignments", "students", "audit"]
HEAD = frozen()
TAG = sys.argv[4] if len(sys.argv) > 4 else HEAD
before = sha()
print(f"mode={mode} head={HEAD}", flush=True)
try:
    if mode == "repro":
        # The new tests over the base's production code.
        for f in PROD:
            old = subprocess.run(
                ["git", "show", f"{BASE}:{f}"], cwd=W, capture_output=True, text=True, check=True
            ).stdout
            (W / f).write_text(old)
        clear_pycache()
        test(NEW, f"repro_{TAG}_tests_over_{BASE}_code", no_bytecode=True)
    elif mode == "modules":
        test(MODULES, f"a_modules_and_guards_{TAG}")
    elif mode == "mutants":
        clear_pycache()
        rc, _ = test(MUTANT_MODULES, "mutants_baseline", "h38n3_settings_mut", True)
        assert rc == 0, "the mutants' baseline is not green"
        rows = []
        for name, (f, a, b) in MUTANTS.items():
            text = (W / f).read_text()
            assert text.count(a) == 1, (name, text.count(a))
            clear_pycache()
            (W / f).write_text(text.replace(a, b))
            try:
                rc, keep = test(
                    MUTANT_MODULES, "mutant_" + name.split()[0], "h38n3_settings_mut", True
                )
            finally:
                (W / f).write_text(text)
                clear_pycache()
            restored = sha() == before
            killers = [line.split(" (")[0] for line in keep if line.startswith(("FAIL:", "ERROR:"))]
            verdict = "KILLED" if rc else "SURVIVED"
            rows.append(f"{HEAD}\t{name}\t{verdict}\trestored={restored}\t{'; '.join(killers)}")
            print(name, verdict, f"restored={restored}", flush=True)
            assert restored, "restore failed"
        (LOGS / "mutation_results.tsv").write_text("\n".join(rows) + "\n")
        killed = sum("\tKILLED\t" in r for r in rows)
        print(f"{killed}/{len(rows)} killed", flush=True)
    elif mode == "regression":
        cmd = WRAP + [
            "flock", os.path.expanduser("~/.machine-fullsuite.lock"),
            "timeout", "-k", "60", "1800", PY, "manage.py", "test",
            "--settings=settings_worktree", "--noinput", "-v", "2", "--parallel", "2", *APPS,
        ]  # fmt: skip
        log = LOGS / ("c_regression_" + "_".join(APPS) + "_" + TAG + ".log")
        started = time.time()
        with log.open("w") as out:
            env = {
                k: v for k, v in os.environ.items()
                if not k.startswith("RACE_COST_") and k != "AUDIT_BENCH"
            }  # fmt: skip
            env["EXEMPT_EMAIL_DOMAINS"] = ""
            proc = subprocess.Popen(
                cmd, cwd=W, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True, env=env
            )
            for line in proc.stdout:
                out.write(time.strftime("%H:%M:%S ") + line)
                out.flush()
            rc = proc.wait()
            out.write(f"exit={rc} wall_s={int(time.time() - started)}\n")
        keep = [
            line for line in log.read_text().splitlines()
            if line[9:].startswith(("Ran ", "OK", "FAILED", "FAIL:", "ERROR:"))
        ]  # fmt: skip
        print(f"=== regression: exit {rc}\n" + "\n".join(keep), flush=True)
finally:
    if mode == "repro":
        subprocess.run(["git", "checkout", "--", *PROD], cwd=W)
        clear_pycache()
    assert sha() == before, "restore failed"
    print("restored ok", flush=True)
