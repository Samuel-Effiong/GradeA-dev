to E1a's scope). One planted bug per mechanism, the guard/parity/snapshot
suites run, the file restored from a saved copy and its sha256 checked.

Usage: python mutants.py <worktree> <python> <logdir> [ids,comma,separated]
"""

import hashlib
import subprocess
import sys
from pathlib import Path

WT, PY, LOGDIR = Path(sys.argv[1]), sys.argv[2], Path(sys.argv[3])
ONLY = set(sys.argv[4].split(",")) if len(sys.argv) > 4 else None
LOGDIR.mkdir(parents=True, exist_ok=True)

GS = "classrooms/grade_service.py"
V = "dashboard/views.py"

    "classrooms.tests_e1_grade_guard",

# (id, description, file, old, new, extra test labels, test that must fail)

    ("M1", 'a second plain mean, Avg("submissions__score_percentage"), in a dashboard view',

def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


summary = []
for mid, desc, rel, old, new, extra, must_fail in MUTANTS:
    if ONLY and mid not in ONLY:
        continue
    path = WT / rel
    saved = path.read_bytes()
    before = sha(path)
    text = saved.decode()
    if old is None:
        mutated = text + new
    else:
        assert text.count(old) == 1, f"{mid}: {old!r} found {text.count(old)} times"
        mutated = text.replace(old, new)
    path.write_text(mutated)
    try:
        proc = subprocess.run(
            [PY, "-W", "ignore", "manage.py", "test", *E1, *extra,
             "--settings=settings_worktree", "--noinput", "--keepdb", "-v", "2"],
            cwd=WT, capture_output=True, text=True,
        )
    finally:
        path.write_bytes(saved)
    after = sha(path)
    assert before == after, f"{mid}: restore failed"
    out = proc.stdout + proc.stderr
    (LOGDIR / f"{mid}.log").write_text(out)
    # The "FAIL:"/"ERROR:" headers, not "... FAIL" result lines: a test that
    # logs while it runs splits its result line, and the result is missed.
    red = sorted({
        line.split()[1]
        for line in out.splitlines()
        if line.startswith(("FAIL: ", "ERROR: "))
    })
    ran = next((l for l in out.splitlines() if l.startswith("Ran ")), "?")
    killed = must_fail in red
    line = (f"{mid:4} {'KILLED' if killed else 'SURVIVED'}  {desc}\n"
            f"     planted in {rel}; must fail: {must_fail}; {ran}\n"
            f"     red: {' '.join(red) or '(none)'}\n"
            f"     sha256 before=after={after}")
    print(line, flush=True)
    summary.append(line)
(LOGDIR / "SUMMARY.txt").write_text("\n".join(summary) + "\n")
