the BASE code, run the snapshot + live-oracle tests, record which go red,
restore by sha256. Usage: python coverage_probe.py <worktree> <python> <logdir>"""

TESTS = ["classrooms.tests_e1_snapshots", "classrooms.tests_e1_grade_parity.LiveSitesMatchLegacyTest"]


def sha(p):
    return hashlib.sha256(p.read_bytes()).hexdigest()


def replace_nth(text, old, new, n):
    idx = -1
    for _ in range(n):
        idx = text.find(old, idx + 1)
        if idx < 0:
            raise SystemExit(f"occurrence {n} of {old!r} not found")
    return text[:idx] + new + text[idx + len(old):]


summary = []
for sid, rel, old, new, occ in SITES:
    if ONLY and sid not in ONLY:
        continue
    path = WT / rel
    original = path.read_bytes()
    before = sha(path)
    path.write_text(replace_nth(original.decode(), old, new, occ))
    try:
        proc = subprocess.run(
            [PY, "-W", "ignore", "manage.py", "test", *TESTS,
             "--settings=settings_worktree", "--noinput", "--keepdb"],
            cwd=WT, capture_output=True, text=True,
        )
    finally:
        path.write_bytes(original)
    after = sha(path)
    assert before == after, f"{sid}: restore failed"
    out = proc.stdout + proc.stderr
    (LOGDIR / f"{sid}.log").write_text(out)
    red = sorted({line.split()[1] for line in out.splitlines() if line.startswith(("FAIL:", "ERROR:"))})
    ran = next((l for l in out.splitlines() if l.startswith("Ran ")), "?")
    verdict = "RED" if red else "green (UNCOVERED)"
    line = f"{sid:11} {rel}:{occ}  {verdict}  {ran}  {' '.join(red)}  sha256-restored={after[:12]}"
    print(line, flush=True)
    summary.append(line)
(LOGDIR / "SUMMARY.txt").write_text("\n".join(summary) + "\n")
