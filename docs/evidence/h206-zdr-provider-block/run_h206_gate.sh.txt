#!/bin/bash
# H-206 zero retention (beta line). Usage: bash run_h206_gate.sh <tip> 1
# One grant (6G), inside ONE outer inhibit:
#   systemd-inhibit --what=idle:sleep:handle-lid-switch --who=GAP --why=GAP-test-run --mode=block bash <script> <tip> 1
# 0 the red run: the module as at the tip against the code as at the base (services.py and settings.py taken back; own DB),
#   RULE 22: each new red test is red for its WRITTEN reason (a fragment looked for in its own FAIL/ERROR block);
# 1a makemigrations --check; 1b the seam by program (the diff of ai_processor/services.py removes exactly the old comment
#   and the one literal provider block; nothing else is removed); 1 the module + near modules (everything that patches
#   __ai_model or the SDK) + guards; 2 the 14 mutants (rule 17; own DB).
# The regression is 0b's one full run on the frozen tip; it is not in this script.
# A trap on EXIT restores the tree. RULE 18: every run writes straight to a file, stdin from /dev/null, nothing piped.
set -u
TIP=$1; STEP=${2:-1}; BASE=fafed57f
cd /home/bond-servant-in-training/Documents/Projects/Grade-Automator-Plus-zdr-provider-block || exit 1
source /home/bond-servant-in-training/Documents/Virtualenvs/AutoGrader_env/bin/activate
E=docs/evidence/h206-zdr-provider-block
CAP="systemd-run --user --scope -q -p MemoryMax=6G -p MemorySwapMax=0"
export PYTHONFAULTHANDLER=1
export EXEMPT_EMAIL_DOMAINS=
[ "$(git rev-parse HEAD)" = "$(git rev-parse $TIP)" ] || { echo "not at $TIP"; exit 1; }
git diff --quiet HEAD || { echo "tree not clean"; exit 1; }
[ "$STEP" = 1 ] || { echo "only step 1 exists"; exit 1; }
mkdir -p $E
CHANGED="ai_processor/services.py AutoGrader/settings.py"
restore_tree() { git checkout HEAD -- $CHANGED 2>/dev/null; find ai_processor -name __pycache__ -type d -prune -exec rm -rf {} + 2>/dev/null; }
trap restore_tree EXIT
summary() { grep -E "^(Ran |OK|FAILED \(|(FAIL|ERROR): )" "$1" | cut -c1-200 | head -${2:-12}; }
dropmut() { python manage.py shell --settings=settings_worktree_mut -c "
from django.conf import settings
from django.db import connection
name = settings.DATABASES['default']['TEST']['NAME']
with connection._nodb_cursor() as c:
    c.execute('DROP DATABASE IF EXISTS \"%s\"' % name)
print('dropped', name)" < /dev/null > $E/.dropmut.out 2>&1; tail -1 $E/.dropmut.out; rm -f $E/.dropmut.out; }
OWN="ai_processor.tests_provider_block"
NEAR="billing.tests.test_execute_graded_task billing.tests.test_refusal_handling billing.tests.test_refusal_handling_gates billing.tests.test_h38_part2_removed_teacher_routes ai_processor.tests_grading_pipeline ai_processor.tests_grading_completeness ai_processor.tests_grading_hardening ai_processor.tests_superadmin_unmetered_both_flags assignments.tests_upload_batch_billing assignments.tests_upload_task_retry_policy dashboard.tests_custom_ai_prompt ai_processor.tests_second_opinion_pipeline ai_processor.tests_second_opinion_selection"
GUARDS="AutoGrader.tests_cache_bespoke_1114 AutoGrader.tests_no_wildcard_invalidation AutoGrader.tests_cache_invalidation_coverage AutoGrader.tests_migration_rollback_defaults AutoGrader.tests_migration_safety_check AutoGrader.tests_redis_test_isolation AutoGrader.tests_beat_health AutoGrader.tests_beat_locks AutoGrader.tests_management_commands_are_commands AutoGrader.tests_error_messages AutoGrader.tests_no_pii_in_logs AutoGrader.tests_log_scrubbing AutoGrader.tests_sentry_scrubbing AutoGrader.tests_redis_hygiene_databases AutoGrader.tests_redis_hygiene AutoGrader.tests_patient_test_stream AutoGrader.tests_pool_worker_sigterm AutoGrader.tests_student_feedback_guard AutoGrader.tests_submission_audience_guard AutoGrader.tests_student_classmates_guard AutoGrader.tests_network_guard"
echo "load at start: $(cat /proc/loadavg)"
echo "=== 0. the red run: $OWN against the code as at $BASE (own DB) $(date +%T)"
git checkout $BASE -- $CHANGED
find ai_processor -name __pycache__ -type d -prune -exec rm -rf {} +
eval "PYTHONDONTWRITEBYTECODE=1 $CAP nice -n 10 timeout -k 60 1800 python manage.py test --noinput --verbosity 2 --settings=settings_worktree_mut $OWN" < /dev/null > $E/red_run_code_as_at_base.txt 2>&1
RED_EXIT=$?
echo "red run exit=$RED_EXIT (non-zero expected)"
RF=$E/red_run_code_as_at_base.txt
RC=/home/bond-servant-in-training/Documents/Projects/GAP-builder-scripts/reason_check.py
RED_OK=1
[ "$RED_EXIT" -ne 0 ] || RED_OK=0
grep -qE "^OK" $RF && RED_OK=0
grep -qE "^Ran [0-9]+ tests? in" $RF || { echo "red run: no Ran line"; RED_OK=0; }
grep -qE "unittest.loader._FailedTest" $RF && { echo "red run: the module failed to load"; RED_OK=0; }
# RULE 22: each test below is red for its WRITTEN reason (a fragment written beforehand, looked for in its own block).
# Tests that PASS at the base by design (guards that hold on the code as it is): the_rest_of_the_request_is_unchanged,
# the_override_and_the_fallback_list_do_not_change_the_block, the_only_create_call_is_the_one_in_ai_model,
# every_file_..._is_on_the_named_list, every_listed_file_..._carries_a_mark_and_a_reason,
# every_listed_file_other_than_services_has_zero_create_calls, the_provider_host_appears_only_in_listed_files
r() { python -I $RC $RF "$1" "$2" || RED_OK=0; }
r test_the_provider_block_that_reaches_the_sdk_is_exact_on_every_path "'zdr': True"
r test_zdr_is_the_boolean_true_not_a_string "None is not True"
r test_zdr_is_inside_the_provider_object_not_the_top_level_body "'zdr' not found in"
r test_the_default_is_on "AI_PROVIDER_ZDR_ENABLED"
r test_the_settings_module_reads_the_variable_with_default_true "AI_PROVIDER_ZDR_ENABLED"
r test_with_the_switch_off_the_block_is_exactly_data_collection_deny "has no attribute '_provider_block'"
r test_the_switch_is_read_at_each_call_not_once_at_import "'zdr': True"
r test_it_exists_and_returns_the_block "has no attribute '_provider_block'"
r test_it_returns_a_new_dict_each_call "has no attribute '_provider_block'"
r test_a_change_to_a_sent_block_cannot_reach_the_next_request "'zdr': True"
r test_the_helper_is_defined_exactly_once_at_module_level "0 != 1"
r test_the_request_takes_the_block_from_the_helper "0 != 1"
r test_the_data_collection_literal_exists_only_inside_the_helper "ai_processor/services.py:"
r test_the_old_claim_that_deny_refuses_providers_that_retain_is_gone "unexpectedly found"
r test_the_comment_says_deny_filters_suppliers_that_train_and_zdr_those_that_retain "suppliers that train on data"
if [ "$RED_OK" != 1 ]; then
    echo "STOP (exit 4): the red run is not what was written"
    git checkout HEAD -- $CHANGED
    find ai_processor -name __pycache__ -type d -prune -exec rm -rf {} +
    exit 4
fi
echo "red run is as written: non-zero exit, no OK line, the module loaded, fifteen tests red for their written reasons"
git checkout HEAD -- $CHANGED
find ai_processor -name __pycache__ -type d -prune -exec rm -rf {} +
git diff --quiet HEAD -- . ":!$E" && [ -z "$(git status --porcelain -- . ":!$E" | grep -v settings_worktree_mut.py)" ] && echo "source restored" || { echo "STOP: not restored"; exit 3; }
grep -E "^(Ran |OK|FAILED \\()" $RF | cut -c1-200
dropmut
echo "=== 1a. makemigrations --check $(date +%T)"
python manage.py makemigrations --check --dry-run --settings=settings_worktree < /dev/null > $E/makemigrations_check.txt 2>&1; MM=$?; echo "makemigrations exit=$MM (0 = no changes)"; tail -1 $E/makemigrations_check.txt
[ "$MM" = 0 ] || { echo "STOP: makemigrations reports changes"; exit 2; }
echo "=== 1b. the seam: what the diff of services.py removes, by program"
git diff $BASE HEAD -U0 -- ai_processor/services.py | grep -E '^-[^-]' > $E/seam_removed_lines.txt
cat $E/seam_removed_lines.txt
[ "$(wc -l < $E/seam_removed_lines.txt)" = 4 ] || { echo "STOP: the diff removes other than the old comment (3 lines) and the one literal block (1 line)"; exit 2; }
[ "$(grep -c 'provider.*data_collection.*deny' $E/seam_removed_lines.txt)" = 1 ] || { echo "STOP: the one literal provider block is not what is removed"; exit 2; }
echo "=== 1. own module + near modules + guards at $(git rev-parse --short=8 HEAD) $(date +%T)"
eval "$CAP nice -n 10 timeout -k 60 3500 python manage.py test --noinput --verbosity 2 --settings=settings_worktree $OWN $NEAR $GUARDS" < /dev/null > $E/modules_and_guards.txt 2>&1
echo "exit=$? $(date +%T)"; summary $E/modules_and_guards.txt
if grep -qE "^FAILED \\(" $E/modules_and_guards.txt || ! grep -qE "^OK" $E/modules_and_guards.txt; then echo "STOP: not green"; exit 2; fi
echo "=== 2. mutants: 14 (own DB, rules 17 and 18) $(date +%T)"
eval "MUT_SETTINGS=settings_worktree_mut MUT_RESULTS=$E/mutation_results.json MUT_LOGS=$E/mutant_logs $CAP nice -n 10 timeout -k 60 3000 python $E/mutate.py" < /dev/null > $E/mutation_log.txt 2>&1
echo "mutants exit=$? $(date +%T)"; cut -c1-260 $E/mutation_log.txt
git diff --quiet HEAD -- . ":!$E" && echo "source clean after mutants" || echo "STOP: source not clean after mutants"
dropmut
echo "load at end: $(cat /proc/loadavg)"
