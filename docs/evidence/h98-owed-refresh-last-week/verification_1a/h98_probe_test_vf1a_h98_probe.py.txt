"""1a's verification probes for H-98. Not committed.

H-98's own tests use hand-built rows and the pure function. These drive
REAL rows: a teacher enrolled through the service in the middle of a cycle,
refreshed by the real daily task, and renewed by the real renewals.

W1: an outage that runs to the cycle's end, with the teacher's last monthly
    point in the last week: the renewal reports it. The stored anchor and the
    due time, read back from the database, are exactly on one chain (a
    clamped month end too). A longer outage counts the earlier points too.
W2: the same rows with Beat healthy: nothing is reported. Swept over the
    distance from the last point to the end, on both sides of the 7-day line
    and of the one-day margin.
W3: the one-day margin against a run that starts late: on time, nothing is
    reported; 45 seconds late, in the narrowest case, the point is not
    served and the renewal reports it.
W4: an offline renewal BEFORE the cycle's end (the count then runs to the
    renewal's moment): a point still in the future, or due less than a day
    ago, is not reported; one due more than a day ago and unserved is.
W5: the author's suggestion, on a real row: a stored anchor that is not the
    one in use, and a due time near a point but not on it, report nothing.
"""

import logging
from contextlib import contextmanager
from datetime import datetime, timedelta
from datetime import timezone as dt_timezone

from dateutil.relativedelta import relativedelta

from billing.license_service import LicenseSubscriptionService
from billing.models import LicenseSubscription, SchoolCreditAllocation
from billing.refresh_timing import is_anchor_point
from billing.tasks import process_license_monthly_credit_refreshes
from billing.tests.test_licence_grant_anchor import LicenceClockTestCase

UTC = dt_timezone.utc
START = datetime(2026, 3, 2, 1, 0, tzinfo=UTC)
END = START + relativedelta(years=1)  # 2 March 2027, 01:00
DAY = timedelta(days=1)
HOUR = timedelta(hours=1)


class Owed(logging.Handler):
    def __init__(self):
        super().__init__(level=logging.ERROR)
        self.lines = []

    def emit(self, record):
        message = record.getMessage()
        if "owed" in message:
            self.lines.append(message)


@contextmanager
def owed_lines():
    handler = Owed()
    logger = logging.getLogger("billing.license_service")
    logger.addHandler(handler)
    try:
        yield handler.lines
    finally:
        logger.removeHandler(handler)


class H98Probe(LicenceClockTestCase):
    start = START
    end = END

    def setUp(self):
        super().setUp()
        self.licence_sub = self.licence(self.start)

    def enrol_with_last_point(self, before_end):
        """A teacher whose 11th monthly point lies `before_end` before the
        cycle's end (give or take the clock's few milliseconds)."""
        allocation = self.enrol(
            self.licence_sub, self.end - before_end - relativedelta(months=11)
        )
        allocation.refresh_from_db()
        point = allocation.grant_anchor_at + relativedelta(months=11)
        return allocation, point

    def renew_by_task(self, at):
        self.clock.moment = at
        self.licence_sub.refresh_from_db()
        with owed_lines() as lines:
            LicenseSubscriptionService.process_license_renewal(self.licence_sub)
        self.licence_sub.refresh_from_db()
        self.assertGreater(self.licence_sub.billing_cycle_end, at, "it did not renew")
        return lines

    def renew_offline(self, at):
        self.clock.moment = at
        self.licence_sub.refresh_from_db()
        with owed_lines() as lines:
            LicenseSubscriptionService.process_offline_renewal(
                self.licence_sub,
                performed_by=self.admin,
                new_billing_cycle_end=at + relativedelta(months=12),
            )
        return lines

    def assertOneOwed(self, lines, allocation, count):
        self.assertEqual(len(lines), 1, lines)
        self.assertIn(f"is owed {count} monthly refresh", lines[0])
        for part in (self.licence_sub.id, allocation.id, allocation.user_id):
            self.assertIn(str(part), lines[0])
        self.assertNotIn("@", lines[0])

    # W1 -----------------------------------------------------------------
    def w1(self, name, enrolled_at, outage_from_month, owed):
        allocation = self.enrol(self.licence_sub, enrolled_at)
        allocation.refresh_from_db()
        anchor = allocation.grant_anchor_at
        self.drive(allocation, until=anchor + relativedelta(months=outage_from_month))
        allocation.refresh_from_db()
        due = allocation.next_credit_grant_at

        # The real rows are exactly on one chain, after eight or ten real
        # refreshes and the database's round trip.
        self.assertEqual(allocation.grant_anchor_at, anchor)
        self.assertEqual(due, anchor + relativedelta(months=outage_from_month))
        self.assertTrue(is_anchor_point(allocation.grant_anchor_at, due))
        last = anchor + relativedelta(months=11)
        self.assertLess(self.end - last, timedelta(days=7))
        self.assertGreaterEqual(self.end - last, DAY)

        lines = self.renew_by_task(self.end + HOUR)

        self.assertOneOwed(lines, allocation, owed)
        print(
            f"\nVF1A_H98_W1 real rows, outage to the end, anchored on {name}: "
            f"last point {last:%d %b %H:%M}, {self.end - last} before the end "
            f"-> owed {owed}"
        )

    # W2 -----------------------------------------------------------------
    def w2(self, label, before_end, outage, owed_after_outage):
        allocation, point = self.enrol_with_last_point(before_end)
        self.drive(allocation, until=point if outage else self.end)
        allocation.refresh_from_db()
        served = allocation.next_credit_grant_at >= self.end

        lines = self.renew_by_task(self.end + HOUR)

        if not outage:
            # Beat healthy: never a report, whether a run fell in or not.
            self.assertEqual(lines, [])
        else:
            self.assertFalse(served)
            if owed_after_outage:
                self.assertOneOwed(lines, allocation, 1)
            else:
                self.assertEqual(lines, [])
        print(
            f"\nVF1A_H98_W2 last point {label} before the end, "
            f"{'outage' if outage else 'healthy'}: served={served}, "
            f"owed lines={len(lines)}"
        )

    # W4 -----------------------------------------------------------------
    def w4(self, label, outage, after_point, owed):
        allocation, point = self.enrol_with_last_point(timedelta(days=3))
        renew_at = point + after_point
        self.assertLess(renew_at, self.end)
        self.drive(allocation, until=point if outage else renew_at)

        lines = self.renew_offline(renew_at)

        if owed:
            self.assertOneOwed(lines, allocation, owed)
        else:
            self.assertEqual(lines, [])
        print(f"\nVF1A_H98_W4 offline renewal before the end; {label}: owed lines={len(lines)}")

    # W5 -----------------------------------------------------------------
    def test_w5_a_stored_anchor_not_in_use_and_a_near_miss(self):
        allocation, point = self.enrol_with_last_point(timedelta(days=3))
        self.drive(allocation, until=point)
        rows = SchoolCreditAllocation.objects.filter(pk=allocation.pk)

        # Near the stored anchor's point, not on it (something moved it).
        rows.update(next_credit_grant_at=point - timedelta(hours=2))
        near = self.renew_offline(point + timedelta(hours=30))
        self.assertEqual(near, [])

        # A stored anchor on another rhythm: the due time becomes the
        # anchor in use, and is on its own chain, but not the stored one.
        self.licence_sub = LicenseSubscription.objects.get(pk=self.licence_sub.pk)
        end = self.licence_sub.billing_cycle_end
        due = end - timedelta(days=3)
        rows.update(
            next_credit_grant_at=due,
            grant_anchor_at=due - relativedelta(months=6) - timedelta(days=15),
        )
        stale = self.renew_offline(end + HOUR)
        self.assertEqual(stale, [])
        print(
            "\nVF1A_H98_W5 a near miss and a stored anchor not in use: "
            f"owed lines {len(near)} and {len(stale)}"
        )


def _add(prefix, method, cases):
    for n, case in enumerate(cases, 1):
        def test(self, case=case):
            getattr(self, method)(*case)

        setattr(H98Probe, f"test_{prefix}_{n:02d}", test)


_add(
    "w1",
    "w1",
    (
        ("the 27th", datetime(2026, 3, 27, 10, 0, tzinfo=UTC), 11, 1),
        ("the 31st (28 Feb, clamped)", datetime(2026, 3, 31, 10, 0, tzinfo=UTC), 11, 1),
        ("the 26th, out from month 9", datetime(2026, 3, 26, 10, 0, tzinfo=UTC), 9, 3),
    ),
)
_add(
    "w2",
    "w2",
    tuple(
        (label, before_end, outage, owed)
        for label, before_end, owed in (
            ("8 days", timedelta(days=8), 1),
            ("7 days + 1 h", timedelta(days=7, hours=1), 1),
            ("7 days", timedelta(days=7), 1),
            ("7 days - 1 h", timedelta(days=6, hours=23), 1),
            ("3 days", timedelta(days=3), 1),
            ("1 day + 1 h", timedelta(days=1, hours=1), 1),
            ("1 day + 1 min", timedelta(days=1, minutes=1), 1),
            ("1 day - 1 min", timedelta(hours=23, minutes=59), 0),
            ("6 h", timedelta(hours=6), 0),
        )
        for outage in (False, True)
    ),
)
_add(
    "w4",
    "w4",
    (
        ("outage, renewed 1 h BEFORE the point", True, -HOUR, 0),
        ("outage, renewed 23 h after the point", True, timedelta(hours=23), 0),
        ("outage, renewed 25 h after the point", True, timedelta(hours=25), 1),
        ("healthy, renewed 25 h after the point", False, timedelta(hours=25), 0),
    ),
)
W_GENERATED = [name for name in vars(H98Probe) if name.startswith(("test_w1_", "test_w2_", "test_w4_"))]


class H98LateRunProbe(H98Probe):
    """W3. The narrowest case for the margin: the point falls due just
    after one daily run's tolerance, and the cycle ends a day and ten
    seconds later, just after the next run's."""

    start = datetime(2026, 3, 2, 3, 5, 30, tzinfo=UTC)
    end = start + relativedelta(years=1)

    test_w5_a_stored_anchor_not_in_use_and_a_near_miss = None
    locals().update(dict.fromkeys(W_GENERATED))

    def last_run(self, late_by):
        allocation, point = self.enrol_with_last_point(DAY + timedelta(seconds=10))
        self.assertLessEqual(point + DAY, self.end)
        self.drive(allocation, until=point)
        run = self.end.replace(hour=3, minute=0, second=0, microsecond=0)
        # The run of the day before finds nothing due yet.
        self.clock.moment = run - DAY
        process_license_monthly_credit_refreshes()
        allocation.refresh_from_db()
        self.assertEqual(allocation.next_credit_grant_at, point)
        # The last run before the cycle's end.
        self.clock.moment = run + late_by
        process_license_monthly_credit_refreshes()
        allocation.refresh_from_db()
        served = allocation.next_credit_grant_at >= self.end
        return allocation, served, self.renew_by_task(self.end + HOUR)

    def test_w3_the_last_run_on_time(self):
        allocation, served, lines = self.last_run(timedelta(0))
        self.assertTrue(served)
        self.assertEqual(lines, [])
        print("\nVF1A_H98_W3 last run on time: served, owed lines=0")

    def test_w3_the_last_run_45_seconds_late(self):
        allocation, served, lines = self.last_run(timedelta(seconds=45))
        self.assertFalse(served)
        self.assertOneOwed(lines, allocation, 1)
        print(
            "\nVF1A_H98_W3 last run 45 s late: not served (the cycle ends "
            "inside the run's 5-minute tolerance), owed lines=1"
        )
