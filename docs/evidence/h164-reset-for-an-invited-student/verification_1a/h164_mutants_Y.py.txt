def main():
    code = sys.argv[1]
    if code not in MUTANTS:
        print("usage: Y1..Y9")
        sys.exit(2)
    old, new = MUTANTS[code]
    with open(F, encoding="utf-8") as handle:
        source = handle.read()
    if source.count(old) != 1:
        print(f"ABORT {code}: anchor found {source.count(old)} times")
        sys.exit(7)
    mutated = source.replace(old, new)
    ast.parse(mutated)
    if mutated == source:
        print(f"ABORT {code}: equals source")
        sys.exit(7)
    if "--check" in sys.argv:
        print(f"checked {code}: one anchor, mutant parses and differs")
        return
    with open(F, "w", encoding="utf-8") as handle:
        handle.write(mutated)
    print(f"applied {code} to {F}")


if __name__ == "__main__":
    main()
