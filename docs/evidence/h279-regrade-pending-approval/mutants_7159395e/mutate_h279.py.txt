LABELS = [
    "students.tests_h279_regrade_pending_approval",
    "students.tests_h279_regrade_pending_concurrency",
]
out = S / "mutants_h279_7159395e"
out.mkdir(exist_ok=True)
files = sorted({p for edits in M.values() for p, _, _ in edits})
committed = {
    p: subprocess.run(["git", "-C", str(W), "show", f"{COMMIT}:{p}"], capture_output=True, check=True).stdout
    for p in files
}
SHA = {p: hashlib.sha256(b).hexdigest() for p, b in committed.items()}


def sha(p):
    return hashlib.sha256((W / p).read_bytes()).hexdigest()


for p in files:
    assert sha(p) == SHA[p], p
print("committed sha256:", SHA)
for name, edits in M.items():
    if ONLY and name not in ONLY:
        continue
    touched = sorted({p for p, _, _ in edits})
    try:
        for p, old, new in edits:
            src = (W / p).read_text()
            assert src.count(old) == 1, (name, p, src.count(old))
            (W / p).write_text(src.replace(old, new, 1))
        r = subprocess.run(
            [PY, "manage.py", "test", "--noinput", "--settings=settings_worktree", "-v", "2", *LABELS],
            cwd=W, capture_output=True, text=True, timeout=1800,
        )
        log = r.stdout + r.stderr
        rc = r.returncode
    finally:
        for p in touched:
            (W / p).write_bytes(committed[p])
    for p in touched:
        assert sha(p) == SHA[p], "restore failed " + p
    head = "".join(f"{name}: {p}\n- {old!r}\n+ {new!r}\n" for p, old, new in edits)
    suffix = "_rerun" if ONLY else ""
    (out / f"{name}{suffix}.txt").write_text(f"{head}rc={rc}\n\n" + log)
    hits = sorted({l.split(" (")[0] for l in log.splitlines() if l.startswith(("FAIL:", "ERROR:"))})
    tail = [l for l in log.splitlines() if l.startswith(("Ran ", "OK", "FAILED"))]
    print(name, "rc=%d" % rc, "restored ok", tail, flush=True)
    print("  " + "\n  ".join(hits), flush=True)
