#!/bin/bash
# H-222 gate chain on the frozen tip: (r) the row's tests at the base d2ad0405
# (phase2/epic-a: neither expire_bucket nor the clawback locks the wallet), in a
# disposable worktree, expected red on a set and for reasons written beforehand;
# (a) the row's modules, the existing billing/audit modules that call
# expire_bucket, the clawback, the cleanup or the lock-first helper, the cache
# test (rule 20: AutoGrader.tests_cache_bespoke_1114) and the repo-wide guard
# list, serial; (b) the battery of 4 mutants. (c) is c_h222.sh, on its own
# grant.
# Rule 18: every test command writes straight to a file, stdin from
# /dev/null. No wall-clock assertion is in these runs; load is recorded.
# Usage: chain.sh <tip-sha8>
set -euo pipefail
TIP=$1
P=/home/bond-servant-in-training/Documents/Projects
WT=$P/Grade-Automator-Plus-h222-wallet-lock-first
RW=$P/Grade-Automator-Plus-h222-repro
OUT=$P/GAP-d5-runs/h222
E=docs/evidence/h222-wallet-lock-first
export EXEMPT_EMAIL_DOMAINS=
WRAP="systemd-run --user --scope -q -p MemoryMax=6G -p MemorySwapMax=0 nice -n 10 timeout -k 60 1800"
say() { echo "$(date +%H:%M:%S) $*" | tee -a $OUT/chain.status; }
load() { say "$1: loadavg $(cat /proc/loadavg)"; }
cd $WT
[ "$(git rev-parse --short=8 HEAD)" = "$TIP" ] && [ -z "$(git status --porcelain)" ] || { say "ABORT: tree not frozen at $TIP"; exit 9; }
[ ! -e $OUT/r_repro_d2ad0405.log ] && [ ! -e $OUT/a_modules_$TIP.log ] && [ ! -e $OUT/battery_$TIP ] || { say "ABORT: logs of this chain exist already"; exit 9; }
[ ! -e $RW ] || { say "ABORT: $RW exists already"; exit 9; }
BATTERY_RAN=0
tidy() {
  set +e
  cd $WT
  if [ -e $RW ]; then git worktree remove --force $RW; git worktree prune; fi
  if [ $BATTERY_RAN = 1 ] && { [ -e $E/logs ] || [ -e $E/results.tsv ]; }; then
    mkdir -p $OUT/battery_$TIP && mv $E/logs $E/results.tsv $OUT/battery_$TIP/ || say "WARNING: could not move the battery's output"
  fi
  [ -z "$(git status --porcelain)" ] && [ "$(git rev-parse --short=8 HEAD)" = "$TIP" ] || say "WARNING: tree not the frozen tip $TIP at exit"
}
trap tidy EXIT

repro() { # name commit label flag [file to take from the tip]
  local name=$1 commit=$2 label=$3 flag=$4 from_tip=${5:-} rc=0 rc2=0 s
  git worktree add --detach $RW $commit > /dev/null 2>&1
  if [ -n "$from_tip" ]; then for f in $from_tip; do git show $TIP:$f > $RW/$f; done; say "($name) $from_tip taken from $TIP, the rest is $commit"; fi
  ln -s $P/Grade-Automator-Plus/.env $RW/.env
  printf 'from AutoGrader.settings import *  # noqa: F401,F403\nfrom AutoGrader.settings import DATABASES\n\nDATABASES["default"].setdefault("TEST", {})\nDATABASES["default"]["TEST"]["NAME"] = "test_h222_repro"\n' > $RW/settings_worktree.py
  load "($name) start"; s=$(date +%s)
  ( cd $RW && PYTHONDONTWRITEBYTECODE=1 $WRAP python manage.py test $label --settings=settings_worktree --noinput > $OUT/${name}_repro_$commit.log 2>&1 < /dev/null ) || rc=$?
  echo "exit=$rc wall_s=$(( $(date +%s)-s ))" >> $OUT/${name}_repro_$commit.log
  load "($name) end"
  say "($name) exit=$rc (1 is the expected one)"
  git worktree remove --force $RW; git worktree prune
  [ $rc -eq 1 ] || { say "STOP: ($name) did not end red in the ordinary way"; exit 1; }
  python3 $OUT/expected_kills.py $flag $OUT/${name}_repro_$commit.log > $OUT/expected_${name}_$commit.txt 2>&1 || rc2=$?
  say "($name) $(tail -n 1 $OUT/expected_${name}_$commit.txt)"
  [ $rc2 -eq 0 ] || { say "STOP: ($name) the red set differs from the expected one"; exit 1; }
}
repro r d2ad0405 "billing.tests.test_wallet_lock_first_expiry audit.tests_background_attribution.ClawbackRaceTests" --repro "billing/tests/test_wallet_lock_first_expiry.py audit/tests_background_attribution.py"

# (a)
load "(a) start"; s=$(date +%s); rc=0
$WRAP python manage.py test billing.tests.test_wallet_lock_first_expiry audit.tests_background_attribution billing.tests.test_wallet_lock_first billing.tests.test_expire_bucket_race billing.tests.test_monthly_rollover_cleanup_race billing.tests.test_overlapping_run_rechecks billing.tests.test_beat_lock_catch_up billing.tests.test_h38_teacher_removal billing.tests.test_h38_part2_removed_teacher_routes billing.tests.test_license_service billing.tests.test_license_teacher_changes_400 billing.tests.test_s7d_licence_teacher_codes billing.tests.test_mailerlite_sync billing.tests.test_neutral_subscription_refusal billing.tests.test_plan_change_retires_old_bucket billing.tests.test_trial_forfeiture_on_activation billing.tests.test_annual_mid_cycle_grants billing.tests.test_rollover_not_double_counted billing.tests.test_license_overage_offline audit.tests_license_admin_attribution audit.tests_metrics AutoGrader.tests_beat_locks billing.tests.test_refusal_handling billing.tests.test_execute_graded_task AutoGrader.tests_cache_commit_race AutoGrader.tests_cache_bespoke_1114 AutoGrader.tests_student_feedback_guard AutoGrader.tests_migration_safety_check AutoGrader.tests_no_playwright_at_import AutoGrader.tests_no_wildcard_invalidation AutoGrader.tests_cache_invalidation_coverage AutoGrader.tests_migration_rollback_defaults AutoGrader.tests_redis_test_isolation AutoGrader.tests_beat_health AutoGrader.tests_management_commands_are_commands AutoGrader.tests_no_pii_in_logs AutoGrader.tests_log_scrubbing AutoGrader.tests_sentry_scrubbing AutoGrader.tests_redis_hygiene_databases AutoGrader.tests_redis_hygiene AutoGrader.tests_patient_test_stream AutoGrader.tests_pool_worker_sigterm billing.tests.test_logs_carry_no_email billing.tests.test_log_scrubbing_end_to_end --settings=settings_worktree --noinput --verbosity 2 > $OUT/a_modules_$TIP.log 2>&1 < /dev/null || rc=$?
echo "exit=$rc wall_s=$(( $(date +%s)-s ))" >> $OUT/a_modules_$TIP.log
load "(a) end"
say "(a) exit=$rc"; [ $rc -eq 0 ] || { say "STOP: (a) red"; exit 1; }

# (b) the battery
load "(b) start"; s=$(date +%s); rc=0; BATTERY_RAN=1
PYTHONDONTWRITEBYTECODE=1 $WRAP python $E/run_mutants.py $TIP > $OUT/b_mutation_battery_$TIP.log 2>&1 < /dev/null || rc=$?
echo "exit=$rc wall_s=$(( $(date +%s)-s ))" >> $OUT/b_mutation_battery_$TIP.log
load "(b) end"
say "(b) exit=$rc"
tidy; BATTERY_RAN=0
rc2=0; python3 $OUT/expected_kills.py $OUT/battery_$TIP > $OUT/expected_kills_$TIP.txt 2>&1 || rc2=$?
say "expected failing sets: $(tail -n 1 $OUT/expected_kills_$TIP.txt)"
[ $rc -eq 0 ] || { say "STOP: (b) not all killed (exit 124 or 137 is a hang or a cap, reported as such)"; exit 1; }
[ $rc2 -eq 0 ] || { say "STOP: a mutant's failing set differs from the expected one"; exit 1; }
say "H-222 CHAIN END: (c) needs its own GRANT"
