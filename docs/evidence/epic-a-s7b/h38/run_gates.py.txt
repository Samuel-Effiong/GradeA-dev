def test(labels, tag, db_suffix=""):
    cmd = ["nice", "-n", "10", "timeout", "-k", "60", "1800", PY, "manage.py", "test",
           "--settings=settings_worktree", "--noinput", "-v", "2", "--parallel", "1", *labels]
    p = subprocess.run(cmd, cwd=W, capture_output=True, text=True)
    out = p.stdout + p.stderr
    (LOGS / f"{tag}.log").write_text(out)
    lines = [l for l in out.splitlines() if l.startswith(("Ran ", "OK", "FAILED", "FAIL:", "ERROR:"))]
    print(f"=== {tag}: exit {p.returncode}\n" + "\n".join(lines), flush=True)
    return p.returncode

    return {f: hashlib.sha256((W / f).read_bytes()).hexdigest() for f in paths}
