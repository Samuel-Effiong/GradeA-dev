#!/bin/bash
# H-204 provider price sort and no fp4 (beta line). Usage: bash run_h204_gate.sh <tip> 1
# One grant, inside ONE outer inhibit (memory cap = available - 2G, min 6G, read at start; rule 13):
#   systemd-inhibit --what=idle:sleep:handle-lid-switch --who=GAP --why=GAP-test-run --mode=block bash <script> <tip> 1
# 0 the red run: the two test modules as at the tip against the code as at the base (the one production file,
#   ai_processor/services.py, taken back; own DB). RULE 22: each red test is red for its WRITTEN reason (a fragment
#   looked for in every one of its own FAIL/ERROR blocks), and exactly 33 blocks appear;
# 1a makemigrations --check; 1b the seam by program (services.py is the only production file changed; by its syntax
#   tree only _provider_block, PROVIDER_SORT and PROVIDER_QUANTIZATIONS changed as code); 1 the two modules + near
#   modules (every module whose tests reach the real request, the call record, the configs, the second opinion, the
#   benchmark) + guards; 2 the 8 mutants (rule 17; own DB).
# The regression is the batch's one full run on the frozen tip; it is not in this script.
# A trap on EXIT restores the tree. RULE 18: every run writes straight to a file, stdin from /dev/null, nothing piped.
set -u
# BASE = origin/beta after batch 24; the red run takes the one production file back to it.
TIP=$1; STEP=${2:-1}; BASE=0e3028a8
cd /home/bond-servant-in-training/Documents/Projects/Grade-Automator-Plus-provider-price-sort || exit 1
source /home/bond-servant-in-training/Documents/Virtualenvs/AutoGrader_env/bin/activate
E=docs/evidence/h204-provider-price-sort
AVAIL_G=$(awk '/^MemAvailable:/ {print int($2/1048576)}' /proc/meminfo)
MEMG=$((AVAIL_G - 2))
echo "available at start: ${AVAIL_G}G; cap ${MEMG}G"
[ "$MEMG" -ge 6 ] || { echo "STOP (exit 5): cap under 6G; wait and tell the SM"; exit 5; }
# Each run has its own scope unit, gap-h204-<pid>-<part>; the watchdogs stop only those, by exact name.
UNITP=gap-h204-$$
UPART=none
CAP="systemd-run --user --scope -q --unit=\${UNITP}-\${UPART} -p MemoryMax=${MEMG}G -p MemorySwapMax=0"
# Memory watchdog (SM 2026-10-14): available under 1.5 GB stops OUR scope, by its unit name, never by pattern.
( while sleep 10; do
    A=$(awk '/^MemAvailable:/ {print $2}' /proc/meminfo)
    if [ "$A" -lt 1572864 ]; then
        for U in $UNITP-red $UNITP-modules $UNITP-mutants; do
            if systemctl --user is-active -q "$U.scope"; then
                echo "MEMORY STOP $(date +%T): available ${A} kB; stopping $U.scope" >> $E/memory_watchdog.txt
                systemctl --user stop "$U.scope"
            fi
        done
    fi
  done ) &
WATCHDOG=$!
export PYTHONFAULTHANDLER=1
export EXEMPT_EMAIL_DOMAINS=
[ "$(git rev-parse HEAD)" = "$(git rev-parse $TIP)" ] || { echo "not at $TIP"; exit 1; }
git diff --quiet HEAD || { echo "tree not clean"; exit 1; }
[ "$STEP" = 1 ] || { echo "only step 1 exists"; exit 1; }
mkdir -p $E
CHANGED="ai_processor/services.py"
RESTORE="ai_processor/services.py"
restore_tree() { kill $WATCHDOG 2>/dev/null; git checkout HEAD -- $RESTORE 2>/dev/null; find ai_processor -name __pycache__ -type d -prune -exec rm -rf {} + 2>/dev/null; }
trap restore_tree EXIT
summary() { grep -E "^(Ran |OK|FAILED \(|(FAIL|ERROR): )" "$1" | cut -c1-200 | head -${2:-12}; }
dropmut() { python manage.py shell --settings=settings_worktree_mut -c "
from django.conf import settings
from django.db import connection
name = settings.DATABASES['default']['TEST']['NAME']
with connection._nodb_cursor() as c:
    c.execute('DROP DATABASE IF EXISTS \"%s\"' % name)
print('dropped', name)" < /dev/null > $E/.dropmut.out 2>&1; tail -1 $E/.dropmut.out; rm -f $E/.dropmut.out; }
OWN="ai_processor.tests_h204_provider_price_sort ai_processor.tests_provider_block"
NEAR="ai_processor.tests_ai_call_trace ai_processor.tests_reading_config ai_processor.tests_reading_pipeline students.tests_reading_label_routes students.tests_reading_label_static ai_processor.tests_h205_image_fallback ai_processor.tests_h254_second_opinion_haiku_first ai_processor.tests_second_opinion_pipeline ai_processor.tests_second_opinion_selection ai_processor.tests_second_opinion_error_code ai_processor.tests_grading_config ai_processor.tests_grading_pipeline ai_processor.tests_grading_hardening ai_processor.tests_grading_cache ai_processor.tests_call_record ai_processor.tests_call_record_static ai_processor.tests_step_run ai_processor.tests_benchmark_golden ai_processor.tests_grading_benchmark ai_processor.tests_benchmark_request_key ai_processor.tests_extraction_benchmark_golden billing.tests.test_execute_graded_task"
GUARDS="AutoGrader.tests_cache_bespoke_1114 AutoGrader.tests_no_wildcard_invalidation AutoGrader.tests_cache_invalidation_coverage AutoGrader.tests_migration_rollback_defaults AutoGrader.tests_migration_safety_check AutoGrader.tests_redis_test_isolation AutoGrader.tests_beat_health AutoGrader.tests_beat_locks AutoGrader.tests_management_commands_are_commands AutoGrader.tests_error_messages AutoGrader.tests_no_pii_in_logs AutoGrader.tests_log_scrubbing AutoGrader.tests_sentry_scrubbing AutoGrader.tests_redis_hygiene_databases AutoGrader.tests_redis_hygiene AutoGrader.tests_patient_test_stream AutoGrader.tests_pool_worker_sigterm AutoGrader.tests_student_feedback_guard AutoGrader.tests_submission_audience_guard AutoGrader.tests_student_classmates_guard AutoGrader.tests_network_guard"
echo "load at start: $(cat /proc/loadavg)"
echo "=== 0. the red run: $OWN against the code as at $BASE (own DB) $(date +%T)"
git checkout $BASE -- $CHANGED
find ai_processor -name __pycache__ -type d -prune -exec rm -rf {} +
UPART=red
eval "PYTHONDONTWRITEBYTECODE=1 $CAP nice -n 10 timeout -k 60 1800 python manage.py test --noinput --verbosity 2 --settings=settings_worktree_mut $OWN" < /dev/null > $E/red_run_code_as_at_base.txt 2>&1
RED_EXIT=$?
echo "red run exit=$RED_EXIT (non-zero expected)"
RF=$E/red_run_code_as_at_base.txt
RC=/home/bond-servant-in-training/Documents/Projects/GAP-builder-scripts/reason_check.py
RED_OK=1
[ "$RED_EXIT" -ne 0 ] || RED_OK=0
grep -qE "^OK" $RF && RED_OK=0
grep -qE "^Ran [0-9]+ tests? in" $RF || { echo "red run: no Ran line"; RED_OK=0; }
grep -qE "unittest.loader._FailedTest" $RF && { echo "red run: a module failed to load"; RED_OK=0; }
# RULE 22: each test below is red for its WRITTEN reason, in every one of its blocks.
# Green at the base by design: test_with_zdr_off_there_is_no_zdr_key_and_data_collection_is_deny, and every
# tests_provider_block test that does not compare a whole block to EXPECTED or OFF.
r() { python -I $RC $RF "$1" "$2" || RED_OK=0; }
SORT_NONE="None != 'price'"
LIST_NONE="None != ['fp8', 'fp16', 'bf16', 'fp32', 'unknown']"
r test_the_cheapest_supplier_is_tried_first "$SORT_NONE"
r test_the_allowed_quantizations "$LIST_NONE"
r test_unknown_stays_because_grok_and_haiku_publish_none "'unknown' not found in []"
r test_no_fp4_endpoint "None is not true : no quantization list is sent"
r test_with_zdr_on_the_block_keeps_zdr_and_has_both_keys "$SORT_NONE"
r test_each_call_gives_a_new_list "$LIST_NONE"
r test_a_text_grading_request_and_its_deepseek_backup "$SORT_NONE"
r test_a_text_grading_request_with_zdr_on "$SORT_NONE"
r test_an_image_grading_request_and_its_haiku_backup "$SORT_NONE"
r test_the_second_opinion_request "$SORT_NONE"
r test_a_reading_request "$SORT_NONE"
# tests_provider_block: EXPECTED and OFF now carry the two keys; the base sends neither. Two of these fail once per
# request path (8 paths, in subtests).
for t in test_the_provider_block_that_reaches_the_sdk_is_exact_on_every_path test_at_the_default_no_request_carries_zdr_anywhere test_off_sends_no_zdr_and_on_sends_zdr_true_on_every_path test_with_the_switch_off_the_block_is_exactly_data_collection_deny test_the_switch_is_read_at_each_call_not_once_at_import test_it_exists_and_returns_the_block test_it_returns_a_new_dict_each_call test_a_change_to_a_sent_block_cannot_reach_the_next_request; do
    r $t "'sort': 'price'"
done
# 11 + (8 + 1 + 8 + 1 + 1 + 1 + 1 + 1) = 33 blocks, and no other.
[ "$(grep -cE '^(FAIL|ERROR): ' $RF)" = 33 ] || { echo "red run: not exactly 33 FAIL/ERROR blocks ($(grep -cE '^(FAIL|ERROR): ' $RF))"; RED_OK=0; }
[ "$(grep -cE '^FAIL: test_the_provider_block_that_reaches_the_sdk_is_exact_on_every_path ' $RF)" = 8 ] || { echo "red run: the exact-on-every-path test is not red on all 8 paths"; RED_OK=0; }
[ "$(grep -cE '^FAIL: test_off_sends_no_zdr_and_on_sends_zdr_true_on_every_path ' $RF)" = 8 ] || { echo "red run: the off/on test is not red on all 8 paths"; RED_OK=0; }
if [ "$RED_OK" != 1 ]; then
    echo "STOP (exit 4): the red run is not what was written"
    git checkout HEAD -- $CHANGED
    find ai_processor -name __pycache__ -type d -prune -exec rm -rf {} +
    exit 4
fi
echo "red run is as written: non-zero exit, no OK line, the modules loaded, 33 blocks red for their written reasons and no other"
git checkout HEAD -- $CHANGED
find ai_processor -name __pycache__ -type d -prune -exec rm -rf {} +
git diff --quiet HEAD -- . ":!$E" && [ -z "$(git status --porcelain -- . ":!$E" | grep -v settings_worktree_mut.py)" ] && echo "source restored" || { echo "STOP: not restored"; exit 3; }
grep -E "^(Ran |OK|FAILED \\()" $RF | cut -c1-200
dropmut
echo "=== 1a. makemigrations --check $(date +%T)"
python manage.py makemigrations --check --dry-run --settings=settings_worktree < /dev/null > $E/makemigrations_check.txt 2>&1; MM=$?; echo "makemigrations exit=$MM (0 = no changes)"; tail -1 $E/makemigrations_check.txt
[ "$MM" = 0 ] || { echo "STOP: makemigrations reports changes"; exit 2; }
echo "=== 1b. the seam, by program: services.py is the only production file; by its syntax tree only the helper and the two constants change"
git diff --name-only $BASE HEAD > $E/seam_changed_files.txt
cat $E/seam_changed_files.txt
[ "$(grep -vE '^(docs/|ai_processor/tests_(h204_provider_price_sort|provider_block)\.py$)' $E/seam_changed_files.txt)" = "ai_processor/services.py" ] || { echo "STOP: a production file other than services.py changed"; exit 2; }
git show $BASE:ai_processor/services.py > $E/.services_base.py
python -I $E/seam_check.py $E/.services_base.py ai_processor/services.py > $E/seam_code.txt 2>&1; SEAM=$?; rm -f $E/.services_base.py
cat $E/seam_code.txt
[ "$SEAM" = 0 ] || { echo "STOP: the seam check failed"; exit 2; }
echo "=== 1. own modules + near modules + guards at $(git rev-parse --short=8 HEAD) $(date +%T)"
UPART=modules
eval "$CAP nice -n 10 timeout -k 60 1800 python manage.py test --noinput --verbosity 2 --settings=settings_worktree $OWN $NEAR $GUARDS" < /dev/null > $E/modules_and_guards.txt 2>&1
echo "exit=$? $(date +%T)"; summary $E/modules_and_guards.txt
if grep -qE "^FAILED \\(" $E/modules_and_guards.txt || ! grep -qE "^OK" $E/modules_and_guards.txt; then echo "STOP: not green"; exit 2; fi
echo "=== 2. mutants: 8 (own DB, rules 17 and 18) $(date +%T)"
# Time limit 90 s x 8 + 600 = 1320 s, with a 300 s stall watchdog on the newest mutant log (2026-10-14).
UPART=mutants
( sleep 60; while systemctl --user is-active -q "$UNITP-mutants.scope"; do
    NEWEST=$(ls -t $E/mutant_logs/*.txt 2>/dev/null | head -1)
    if [ -n "$NEWEST" ] && [ $(( $(date +%s) - $(stat -c %Y "$NEWEST") )) -gt 300 ]; then
        echo "STALL STOP $(date +%T): $NEWEST silent over 300 s; stopping $UNITP-mutants.scope" >> $E/stall_watchdog.txt
        systemctl --user stop "$UNITP-mutants.scope"; break
    fi
    sleep 15
  done ) &
STALL=$!
eval "MUT_SETTINGS=settings_worktree_mut MUT_RESULTS=$E/mutation_results.json MUT_LOGS=$E/mutant_logs $CAP nice -n 10 timeout -k 60 1320 python $E/mutate.py" < /dev/null > $E/mutation_log.txt 2>&1
echo "mutants exit=$? $(date +%T)"; kill $STALL 2>/dev/null; cut -c1-260 $E/mutation_log.txt
git diff --quiet HEAD -- . ":!$E" && echo "source clean after mutants" || echo "STOP: source not clean after mutants"
dropmut
echo "load at end: $(cat /proc/loadavg)"
