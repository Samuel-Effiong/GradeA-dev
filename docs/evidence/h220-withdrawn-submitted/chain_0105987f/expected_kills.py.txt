W2 = (WS, "test_w2_withdrawn_without_a_submission_is_left_out")
W3 = (WS, "test_w3_a_submission_for_another_assignment_does_not_count")
C1 = (EV, "test_c1_enrolled_pending_and_completed_are_listed_with_or_without_one")
C2 = (EV, "test_c2_the_defined_order_holds_with_a_withdrawn_student_among_them")

O2 = (OR, "test_o2_a_created_at_tie_is_broken_by_the_enrolment_id")

    "R1": {W2, W3, C2, L1, F1},

#: The BASE with the final test modules (taken from the tip). Controls W1, C1,
#: O1, O2, L2 are green by design.
REPRO = {W2, W3, C2, L1, F1}
#: RULE 22: fragments that must be in each red test's OWN failure block at (r),
#: written before the run.
REASON = {
    W2: ("self.assertEqual([row[\"name\"] for row in rows], [self.name_of(present)])", "First list contains 1 additional elements"),
    W3: ("self.assertEqual([row[\"name\"] for row in rows], [self.name_of(present)])", "First list contains 1 additional elements"),
    C2: ("self.assertEqual(len(got), 3)", "4 != 3"),
    L1: ("self.assertEqual(len(got), 2)", "3 != 2"),
    F1: ("self.assertEqual(self.names(), [self.full_name(self.rows[1])])", "First list contains 1 additional elements"),
}
BLOCK = re.compile(
    r"^(?:FAIL|ERROR): (test_\w+) \(([\w.]+)\)\n(.*?)(?=^(?:FAIL|ERROR): |^Ran \d+ tests)",
    re.M | re.S,
)


def wrong_reasons(text):
    blocks = {(p.split(".")[-2], n): body for n, p, body in BLOCK.findall(text)}
    out = []
    for test, fragments in REASON.items():
        if test in blocks:
            missing = [f for f in fragments if f not in blocks[test]]
            if missing:
                out.append(f"{test[1]}: not in its own failure: {missing}")
    return out


LINE = re.compile(r"^(?:FAIL|ERROR): (test_\w+) \(([\w.]+)\)", re.M)


def failing(text):
    return {(path.split(".")[-2], name) for name, path in LINE.findall(text)}


def compare(label, got, expected):
    extra, missing = got - expected, expected - got
    print(
        f"{label}: {len(got)} failing, {'as expected' if got == expected else 'DIFFERS'}"
        + (f"; unexpected: {sorted(extra)}" if extra else "")
        + (f"; expected but passed: {sorted(missing)}" if missing else "")
    )
    return got != expected


if sys.argv[1] == "--repro":
    log = Path(sys.argv[2]).read_text(errors="replace")
    bad = compare("repro", failing(log), REPRO)
    reasons = wrong_reasons(log)
    for line in reasons:
        print("WRONG REASON:", line)
    bad = bad or bool(reasons)
    print("the red set is the expected one, each for its written reason" if not bad else "the red set differs")
else:
    logs = Path(sys.argv[1]) / "logs"
    bad = sum(
        compare(mutant, failing((logs / f"{mutant}.log").read_text(errors="replace")), expected)
        for mutant, expected in EXPECTED.items()
    )
    print("every mutant's failing set is the expected one" if not bad else f"{bad} mutant(s) differ")
sys.exit(1 if bad else 0)
